Lab 45.6: Create All Metric Filters (Script)
bash#!/bin/bash

LOG_GROUP="CloudTrail/security-audit-trail"
NAMESPACE="CloudTrailMetrics"

# Root account usage (already created)
# Skipping...

# Failed console logins
aws logs put-metric-filter \
  --log-group-name "$LOG_GROUP" \
  --filter-name "ConsoleLoginFailures" \
  --filter-pattern '{ $.eventName = "ConsoleLogin" && $.errorMessage = "Failed authentication" }' \
  --metric-transformations \
    metricName=ConsoleLoginFailures,metricNamespace=$NAMESPACE,metricValue=1,defaultValue=0

# Unauthorized API calls
aws logs put-metric-filter \
  --log-group-name "$LOG_GROUP" \
  --filter-name "UnauthorizedAPICalls" \
  --filter-pattern '{ $.errorCode = "AccessDenied" || $.errorCode = "UnauthorizedOperation" }' \
  --metric-transformations \
    metricName=UnauthorizedAPICalls,metricNamespace=$NAMESPACE,metricValue=1,defaultValue=0

# S3 bucket policy changes
aws logs put-metric-filter \
  --log-group-name "$LOG_GROUP" \
  --filter-name "S3BucketPolicyChanges" \
  --filter-pattern '{ $.eventName = "PutBucketPolicy" || $.eventName = "DeleteBucketPolicy" || $.eventName = "PutBucketAcl" }' \
  --metric-transformations \
    metricName=S3BucketPolicyChanges,metricNamespace=$NAMESPACE,metricValue=1,defaultValue=0

# IAM policy changes
aws logs put-metric-filter \
  --log-group-name "$LOG_GROUP" \
  --filter-name "IAMPolicyChanges" \
  --filter-pattern '{ $.eventName = "PutUserPolicy" || $.eventName = "PutGroupPolicy" || $.eventName = "PutRolePolicy" || $.eventName = "CreatePolicy" || $.eventName = "DeletePolicy" }' \
  --metric-transformations \
    metricName=IAMPolicyChanges,metricNamespace=$NAMESPACE,metricValue=1,defaultValue=0

# Security group changes
aws logs put-metric-filter \
  --log-group-name "$LOG_GROUP" \
  --filter-name "SecurityGroupChanges" \
  --filter-pattern '{ $.eventName = "AuthorizeSecurityGroupIngress" || $.eventName = "AuthorizeSecurityGroupEgress" || $.eventName = "RevokeSecurityGroupIngress" || $.eventName = "RevokeSecurityGroupEgress" }' \
  --metric-transformations \
    metricName=SecurityGroupChanges,metricNamespace=$NAMESPACE,metricValue=1,defaultValue=0

# Network gateway changes
aws logs put-metric-filter \
  --log-group-name "$LOG_GROUP" \
  --filter-name "NetworkGatewayChanges" \
  --filter-pattern '{ $.eventName = "CreateInternetGateway" || $.eventName = "AttachInternetGateway" || $.eventName = "DeleteInternetGateway" }' \
  --metric-transformations \
    metricName=NetworkGatewayChanges,metricNamespace=$NAMESPACE,metricValue=1,defaultValue=0

# KMS key deletion
aws logs put-metric-filter \
  --log-group-name "$LOG_GROUP" \
  --filter-name "KMSKeyDeletion" \
  --filter-pattern '{ $.eventName = "DisableKey" || $.eventName = "ScheduleKeyDeletion" }' \
  --metric-transformations \
    metricName=KMSKeyDeletion,metricNamespace=$NAMESPACE,metricValue=1,defaultValue=0

echo "✅ All metric filters created!"

# List filters
aws logs describe-metric-filters \
  --log-group-name "$LOG_GROUP" \
  --query 'metricFilters[].filterName' \
  --output table

sql-- Top 10 users by API call count
SELECT

    COUNT(*) as api_call_count
FROM cloudtrail_logs_table

    AND useridentity.username IS NOT NULL

ORDER BY api_call_count DESC

Query 8: Most Common API Calls
sql-- Top 20 API calls
SELECT
    eventname,
    COUNT(*) as call_count
FROM cloudtrail_logs_table

LIMIT 20;

Query 9: Activity by IP Address
sql-- API calls per source IP
SELECT
    sourceipaddress,
    COUNT(*) as request_count,
    COUNT(DISTINCT eventname) as unique_apis,
    COUNT(DISTINCT useridentity.username) as unique_users
FROM cloudtrail_logs_table
WHERE eventtime > '2026-02-01'
    AND sourceipaddress IS NOT NULL
GROUP BY sourceipaddress
ORDER BY request_count DESC
LIMIT 20;

Query 10: After-Hours Activity
sql-- Activity outside business hours (6 PM - 8 AM UTC)
SELECT
    eventtime,
    eventname,
    useridentity.username,
    sourceipaddress
FROM cloudtrail_logs_table
WHERE eventtime > '2026-02-01'
    AND (
        CAST(date_format(from_iso8601_timestamp(eventtime), '%H') AS INTEGER) >= 18
        OR CAST(date_format(from_iso8601_timestamp(eventtime), '%H') AS INTEGER) <= 8
    )
ORDER BY eventtime DESC
LIMIT 100;

LAB 47: DETECT SUSPICIOUS ACTIVITY
Lab 47.1: Reconnaissance Detection
Pattern: Attacker probing for permissions
sql-- Detect reconnaissance (many failed API calls from same IP)
SELECT
    sourceipaddress,
    useridentity.username,
    COUNT(*) as failed_attempts,
    COUNT(DISTINCT eventname) as unique_apis_tried,
    MIN(eventtime) as first_attempt,
    MAX(eventtime) as last_attempt
FROM cloudtrail_logs_table
WHERE errorcode IN ('AccessDenied', 'UnauthorizedOperation')
    AND eventtime > date_add('day', -1, current_timestamp)
GROUP BY sourceipaddress, useridentity.username
HAVING COUNT(*) > 10
ORDER BY failed_attempts DESC;
Red flags:

Many failed attempts from same IP
Testing multiple different APIs
Short time window (burst)

Lab 47.3: Data Exfiltration Detection
sql-- Detect potential data exfiltration
SELECT
    eventtime,
    eventname,
    useridentity.username,
    sourceipaddress,
    requestparameters,
    COUNT(*) OVER (
        PARTITION BY useridentity.username, DATE(eventtime)
        ORDER BY eventtime
    ) as daily_count
FROM cloudtrail_logs_table
WHERE eventname IN (
    'GetObject',
    'CopyObject',
    'CreateSnapshot',
    'CopySnapshot',
    'CreateDBSnapshot',
    'CopyDBSnapshot',
    'DownloadDBLogFilePortion',
    'GetQueryResults'
)
AND eventtime > date_add('day', -1, current_timestamp)
ORDER BY eventtime DESC
LIMIT 100;
Red flags:

Unusual volume of GetObject calls
Off-hours downloads
New IP addresses
Copying snapshots to external accounts


Lab 47.4: Persistence Mechanism Detection
sql-- Detect attackers establishing persistence
SELECT

    useridentity.username,
    eventname,

Lab 47.5: Crypto-Mining Detection
sql-- Detect crypto-mining (large/expensive instance launches)
SELECT
    eventtime,
    useridentity.username,
    eventname,
    sourceipaddress,
    requestparameters
FROM cloudtrail_logs_table
WHERE eventname = 'RunInstances'
    AND (
        requestparameters LIKE '%p3.%'  -- GPU instances
        OR requestparameters LIKE '%p4.%'
        OR requestparameters LIKE '%g4.%'
        OR requestparameters LIKE '%c5.24xlarge%'  -- Large CPU
    )
AND eventtime > date_add('day', -7, current_timestamp)
ORDER BY eventtime DESC;
```

**Red flags:**
- GPU instances (p3, g4)
- Very large instances
- Unusual regions
- New IAM users launching instances
```
