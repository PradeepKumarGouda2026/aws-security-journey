Replace code with:
pythonimport json
import boto3
import os

ec2 = boto3.client('ec2')
sns = boto3.client('sns')

# Replace with your isolation security group ID
ISOLATION_SG = 'sg-xxxxxxxxx'  # Your isolation-sg ID
SNS_TOPIC_ARN = 'arn:aws:sns:us-east-1:ACCOUNT:GuardDutyAlerts'

def lambda_handler(event, context):
    print("Received GuardDuty finding:")
    print(json.dumps(event, indent=2))
    
    # Extract finding details
    finding = event['detail']
    finding_type = finding.get('type', 'Unknown')
    severity = finding.get('severity', 0)
    
    # Only act on high severity findings
    if severity < 7.0:
        print(f"Severity {severity} below threshold, no action taken")
        return {'statusCode': 200, 'body': 'Low severity, ignored'}
    
    # Get instance ID from finding
    instance_id = None
    if 'resource' in finding:
        resource = finding['resource']
        if resource.get('resourceType') == 'Instance':
            instance_details = resource.get('instanceDetails', {})
            instance_id = instance_details.get('instanceId')
    
    if not instance_id:
        print("No instance ID found in finding")
        return {'statusCode': 200, 'body': 'No instance to isolate'}
    
    print(f"Isolating instance: {instance_id}")
    
    try:
        # Get current security groups
        response = ec2.describe_instances(InstanceIds=[instance_id])
        current_sgs = response['Reservations'][0]['Instances'][0]['SecurityGroups']
        current_sg_ids = [sg['GroupId'] for sg in current_sgs]
        
        # Replace security groups with isolation SG
        ec2.modify_instance_attribute(
            InstanceId=instance_id,
            Groups=[ISOLATION_SG]
        )
        
        message = f"""
GuardDuty Automated Response Executed

Finding Type: {finding_type}
Severity: {severity}
Instance ID: {instance_id}

ACTION TAKEN:
✅ Instance isolated (security group changed to {ISOLATION_SG})
✅ All network traffic blocked

Previous Security Groups: {', '.join(current_sg_ids)}

NEXT STEPS:
1. Take snapshot of instance for forensics
2. Investigate how compromise occurred
3. Review CloudTrail logs for instance
4. Terminate instance after investigation
5. Launch clean replacement

Instance is now quarantined and cannot communicate.
        """
        
        # Send SNS notification
        sns.publish(
            TopicArn=SNS_TOPIC_ARN,
            Subject=f'🚨 Instance {instance_id} Isolated by GuardDuty',
            Message=message
        )
        
        print(f"Successfully isolated instance {instance_id}")
        
        return {
            'statusCode': 200,
            'body': json.dumps({
                'action': 'isolated',
                'instance': instance_id,
                'finding_type': finding_type,
                'severity': severity
            })
        }
        
    except Exception as e:
        error_msg = f"Error isolating instance {instance_id}: {str(e)}"
        print(error_msg)
        
        sns.publish(
            TopicArn=SNS_TOPIC_ARN,
            Subject='❌ GuardDuty Auto-Response Failed',
            Message=f"Failed to isolate instance {instance_id}\n\nError: {str(e)}"
        )
        
        return {
            'statusCode': 500,
            'body': json.dumps({'error': str(e)})
        }
