## Prerequisites Check

```bash
#!/bin/bash
# Lab 102 - Prerequisites Check
# Day 37 - AWS Security Specialist Study Plan

echo "================================================"
echo "  Lab 102: AWS WAF & Shield Prerequisites"
echo "================================================"

# Step 1: Verify identity
echo ""
echo "[1] Verifying AWS identity..."
aws sts get-caller-identity --output table

# Step 2: Capture environment
ACCOUNT_ID=$(aws sts get-caller-identity \
  --query Account --output text)
REGION=$(aws configure get region)
CALLER_ARN=$(aws sts get-caller-identity \
  --query Arn --output text)

echo ""
echo "Account ID : $ACCOUNT_ID"
echo "Region     : $REGION"
echo "Caller ARN : $CALLER_ARN"

# Step 3: Check existing WAF Web ACLs
echo ""
echo "[2] Checking existing WAF Web ACLs (regional)..."
aws wafv2 list-web-acls \
  --scope REGIONAL \
  --query 'WebACLs[*].{
    Name:Name,
    Id:Id,
    ARN:ARN
  }' \
  --output table 2>/dev/null || \
  echo "No existing Web ACLs"

# Step 4: Check WAF in us-east-1 (for CloudFront)
echo ""
echo "[3] Checking WAF Web ACLs in us-east-1 (CloudFront)..."
aws wafv2 list-web-acls \
  --scope CLOUDFRONT \
  --region us-east-1 \
  --query 'WebACLs[*].{
    Name:Name,
    Id:Id
  }' \
  --output table 2>/dev/null || \
  echo "No CloudFront Web ACLs"

# Step 5: Check available ALBs
echo ""
echo "[4] Checking available ALBs..."
aws elbv2 describe-load-balancers \
  --query 'LoadBalancers[*].{
    Name:LoadBalancerName,
    ARN:LoadBalancerArn,
    Type:Type,
    Scheme:Scheme
  }' \
  --output table 2>/dev/null || \
  echo "No ALBs found"

# Step 6: Check Shield subscription
echo ""
echo "[5] Checking Shield subscription..."
aws shield describe-subscription \
  --query 'Subscription.{
    StartTime:StartTime,
    EndTime:EndTime,
    AutoRenew:AutoRenew
  }' \
  --output table 2>/dev/null || \
  echo "Shield Advanced not subscribed (Standard active)"

# Step 7: Save environment
cat > /tmp/lab102-env.sh << EOF
export ACCOUNT_ID="$ACCOUNT_ID"
export REGION="$REGION"
export CALLER_ARN="$CALLER_ARN"
EOF

echo ""
echo "✅ Environment saved to /tmp/lab102-env.sh"
echo "================================================"
echo "Prerequisites check complete!"
echo "================================================"
```

## Part 1 — Create IP Sets & Rule Groups

```bash
#!/bin/bash
# Lab 102 - Part 1: Create IP Sets & Rule Groups
# Day 37 - AWS Security Specialist Study Plan

source /tmp/lab102-env.sh

echo "================================================"
echo "  Part 1: Create IP Sets & Rule Groups"
echo "================================================"

# Step 1: Create blocked IPs set
echo ""
echo "[1] Creating blocked IPs set..."
BLOCKED_IPS_ARN=$(aws wafv2 create-ip-set \
  --name "Lab102-BlockedIPs" \
  --scope REGIONAL \
  --ip-address-version IPV4 \
  --addresses \
    "192.168.100.0/24" \
    "10.0.0.1/32" \
    "172.16.0.0/16" \
  --description "Lab102 known bad IP addresses" \
  --tags Key=Lab,Value=Lab102 \
  --query 'Summary.ARN' \
  --output text 2>/dev/null)

if [ ! -z "$BLOCKED_IPS_ARN" ]; then
  echo "✅ Blocked IPs set created"
  echo "   ARN: $BLOCKED_IPS_ARN"
  echo "export BLOCKED_IPS_ARN=\"$BLOCKED_IPS_ARN\"" \
    >> /tmp/lab102-env.sh

  BLOCKED_IPS_ID=$(aws wafv2 list-ip-sets \
    --scope REGIONAL \
    --query \
      'IPSets[?Name==`Lab102-BlockedIPs`].Id' \
    --output text)
  echo "export BLOCKED_IPS_ID=\"$BLOCKED_IPS_ID\"" \
    >> /tmp/lab102-env.sh
fi

# Step 2: Create allowed IPs set (corporate IPs)
echo ""
echo "[2] Creating corporate allowlist IP set..."
ALLOWED_IPS_ARN=$(aws wafv2 create-ip-set \
  --name "Lab102-CorporateIPs" \
  --scope REGIONAL \
  --ip-address-version IPV4 \
  --addresses \
    "203.0.113.0/24" \
    "198.51.100.0/24" \
  --description "Lab102 corporate office IP ranges" \
  --tags Key=Lab,Value=Lab102 \
  --query 'Summary.ARN' \
  --output text 2>/dev/null)

if [ ! -z "$ALLOWED_IPS_ARN" ]; then
  echo "✅ Corporate IPs allowlist created"
  echo "   ARN: $ALLOWED_IPS_ARN"
  echo "export ALLOWED_IPS_ARN=\"$ALLOWED_IPS_ARN\"" \
    >> /tmp/lab102-env.sh

  ALLOWED_IPS_ID=$(aws wafv2 list-ip-sets \
    --scope REGIONAL \
    --query \
      'IPSets[?Name==`Lab102-CorporateIPs`].Id' \
    --output text)
  echo "export ALLOWED_IPS_ID=\"$ALLOWED_IPS_ID\"" \
    >> /tmp/lab102-env.sh
fi

# Step 3: Create regex pattern set for SQL injection
echo ""
echo "[3] Creating regex pattern set for SQLi detection..."
REGEX_ARN=$(aws wafv2 create-regex-pattern-set \
  --name "Lab102-SQLiPatterns" \
  --scope REGIONAL \
  --regular-expression-list \
    'RegexString=(?i)(union|select|insert|update|delete|drop|create|alter|exec|execute)' \
    'RegexString=(?i)(or\s+\d+=\d+|and\s+\d+=\d+)' \
    'RegexString=(?i)(--|\#|\/\*.*\*\/)' \
  --description "Lab102 SQL injection patterns" \
  --tags Key=Lab,Value=Lab102 \
  --query 'Summary.ARN' \
  --output text 2>/dev/null)

if [ ! -z "$REGEX_ARN" ]; then
  echo "✅ SQLi regex pattern set created"
  echo "   ARN: $REGEX_ARN"
  echo "export REGEX_ARN=\"$REGEX_ARN\"" \
    >> /tmp/lab102-env.sh

  REGEX_ID=$(aws wafv2 list-regex-pattern-sets \
    --scope REGIONAL \
    --query \
      'RegexPatternSets[?Name==`Lab102-SQLiPatterns`].Id' \
    --output text)
  echo "export REGEX_ID=\"$REGEX_ID\"" \
    >> /tmp/lab102-env.sh
fi

# Step 4: Create custom rule group
echo ""
echo "[4] Creating custom rule group..."
RULE_GROUP_ARN=$(aws wafv2 create-rule-group \
  --name "Lab102-CustomRuleGroup" \
  --scope REGIONAL \
  --capacity 100 \
  --description "Lab102 custom security rules" \
  --visibility-config \
    SampledRequestsEnabled=true,\
CloudWatchMetricsEnabled=true,\
MetricName=Lab102CustomRuleGroup \
  --rules '[
    {
      "Name": "BlockBadUserAgents",
      "Priority": 1,
      "Statement": {
        "ByteMatchStatement": {
          "SearchString": "sqlmap",
          "FieldToMatch": {"SingleHeader": {"Name": "user-agent"}},
          "TextTransformations": [
            {"Priority": 0, "Type": "LOWERCASE"}
          ],
          "PositionalConstraint": "CONTAINS"
        }
      },
      "Action": {"Block": {}},
      "VisibilityConfig": {
        "SampledRequestsEnabled": true,
        "CloudWatchMetricsEnabled": true,
        "MetricName": "BlockSQLMapAgent"
      }
    },
    {
      "Name": "BlockScanner",
      "Priority": 2,
      "Statement": {
        "OrStatement": {
          "Statements": [
            {
              "ByteMatchStatement": {
                "SearchString": "nikto",
                "FieldToMatch": {
                  "SingleHeader": {"Name": "user-agent"}
                },
                "TextTransformations": [
                  {"Priority": 0, "Type": "LOWERCASE"}
                ],
                "PositionalConstraint": "CONTAINS"
              }
            },
            {
              "ByteMatchStatement": {
                "SearchString": "nessus",
                "FieldToMatch": {
                  "SingleHeader": {"Name": "user-agent"}
                },
                "TextTransformations": [
                  {"Priority": 0, "Type": "LOWERCASE"}
                ],
                "PositionalConstraint": "CONTAINS"
              }
            }
          ]
        }
      },
      "Action": {"Block": {}},
      "VisibilityConfig": {
        "SampledRequestsEnabled": true,
        "CloudWatchMetricsEnabled": true,
        "MetricName": "BlockScanners"
      }
    }
  ]' \
  --tags Key=Lab,Value=Lab102 \
  --query 'Summary.ARN' \
  --output text 2>/dev/null)

if [ ! -z "$RULE_GROUP_ARN" ]; then
  echo "✅ Custom rule group created"
  echo "   ARN: $RULE_GROUP_ARN"
  echo "export RULE_GROUP_ARN=\"$RULE_GROUP_ARN\"" \
    >> /tmp/lab102-env.sh

  RULE_GROUP_ID=$(aws wafv2 list-rule-groups \
    --scope REGIONAL \
    --query \
      'RuleGroups[?Name==`Lab102-CustomRuleGroup`].Id' \
    --output text)
  echo "export RULE_GROUP_ID=\"$RULE_GROUP_ID\"" \
    >> /tmp/lab102-env.sh
fi

# Step 5: List created resources
echo ""
echo "[5] Listing created WAF resources..."
echo ""
echo "IP Sets:"
aws wafv2 list-ip-sets \
  --scope REGIONAL \
  --query 'IPSets[?contains(Name,`Lab102`)].{
    Name:Name,
    Id:Id
  }' \
  --output table

echo ""
echo "Regex Pattern Sets:"
aws wafv2 list-regex-pattern-sets \
  --scope REGIONAL \
  --query \
    'RegexPatternSets[?contains(Name,`Lab102`)].{
      Name:Name,
      Id:Id
    }' \
  --output table

echo ""
echo "Rule Groups:"
aws wafv2 list-rule-groups \
  --scope REGIONAL \
  --query 'RuleGroups[?contains(Name,`Lab102`)].{
    Name:Name,
    Id:Id,
    Capacity:Capacity
  }' \
  --output table

echo ""
echo "================================================"
echo "Part 1 Complete!"
echo "Created:"
echo "  - Blocked IPs set (3 ranges)"
```

## Part 3 — Configure WAF Logging

```bash
#!/bin/bash
# Lab 102 - Part 3: WAF Logging Pipeline
# Day 37 - AWS Security Specialist Study Plan

source /tmp/lab102-env.sh

echo "================================================"
echo "  Part 3: WAF Logging Pipeline"
echo "================================================"

# Step 1: Create S3 bucket for WAF logs
echo ""
echo "[1] Creating S3 bucket for WAF logs..."
WAF_LOG_BUCKET="aws-waf-logs-lab102-${ACCOUNT_ID}"

aws s3 mb s3://$WAF_LOG_BUCKET \
  --region $REGION 2>/dev/null && \
  echo "✅ WAF log bucket created: $WAF_LOG_BUCKET" || \
  echo "⚠️  Bucket may already exist"

echo "export WAF_LOG_BUCKET=\"$WAF_LOG_BUCKET\"" \
  >> /tmp/lab102-env.sh

# Note: WAF log bucket names must start with aws-waf-logs-
echo "   Note: WAF log bucket names must start with 'aws-waf-logs-'"

# Step 2: Configure bucket lifecycle
echo ""
echo "[2] Configuring S3 lifecycle for log retention..."
aws s3api put-bucket-lifecycle-configuration \
  --bucket $WAF_LOG_BUCKET \
  --lifecycle-configuration '{
    "Rules": [
      {
        "ID": "WAFLogRetention",
        "Status": "Enabled",
        "Filter": {"Prefix": ""},
        "Transitions": [
          {
            "Days": 30,
            "StorageClass": "STANDARD_IA"
          },
          {
            "Days": 90,
            "StorageClass": "GLACIER"
          }
        ],
        "Expiration": {
          "Days": 365
        }
      }
    ]
  }' 2>/dev/null && \
  echo "✅ S3 lifecycle configured (30d→IA, 90d→Glacier, 365d→delete)"

# Step 3: Create Kinesis Firehose for WAF logs
echo ""
echo "[3] Creating Kinesis Firehose delivery stream..."

# Create IAM role for Firehose
cat > /tmp/firehose-trust.json << 'EOF'
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Effect": "Allow",
      "Principal": {
        "Service": "firehose.amazonaws.com"
      },
      "Action": "sts:AssumeRole"
    }
  ]
}
EOF

FIREHOSE_ROLE_ARN=$(aws iam create-role \
  --role-name "Lab102-FirehoseRole" \
  --assume-role-policy-document \
    file:///tmp/firehose-trust.json \
  --query 'Role.Arn' \
  --output text 2>/dev/null)

echo "Firehose Role ARN: $FIREHOSE_ROLE_ARN"

# Attach S3 policy to Firehose role
cat > /tmp/firehose-policy.json << EOF
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Effect": "Allow",
      "Action": [
        "s3:PutObject",
        "s3:GetBucketLocation",
        "s3:ListBucket",
        "s3:GetObject"
      ],
      "Resource": [
        "arn:aws:s3:::$WAF_LOG_BUCKET",
        "arn:aws:s3:::$WAF_LOG_BUCKET/*"
      ]
    },
    {
      "Effect": "Allow",
      "Action": [
        "logs:CreateLogGroup",
        "logs:CreateLogStream",
        "logs:PutLogEvents"
      ],
      "Resource": "*"
    }
  ]
}
EOF

aws iam put-role-policy \
  --role-name "Lab102-FirehoseRole" \
  --policy-name "Lab102-FirehoseS3Policy" \
  --policy-document \
    file:///tmp/firehose-policy.json \
  2>/dev/null && echo "✅ Firehose S3 policy attached"

echo "   Waiting 10s for IAM role to propagate..."
sleep 10

# Create Firehose delivery stream
# Note: WAF firehose name must start with aws-waf-logs-
FIREHOSE_ARN=$(aws firehose create-delivery-stream \
  --delivery-stream-name "aws-waf-logs-lab102" \
  --delivery-stream-type DirectPut \
  --s3-destination-configuration \
    RoleARN=$FIREHOSE_ROLE_ARN,\
BucketARN=arn:aws:s3:::$WAF_LOG_BUCKET,\
Prefix=waf-logs/,\
ErrorOutputPrefix=waf-errors/,\
BufferingHints={SizeInMBs=5,IntervalInSeconds=300},\
CompressionFormat=GZIP \
  --query 'DeliveryStreamARN' \
  --output text 2>/dev/null)

if [ ! -z "$FIREHOSE_ARN" ]; then
  echo "✅ Kinesis Firehose created"
  echo "   ARN: $FIREHOSE_ARN"
  echo "export FIREHOSE_ARN=\"$FIREHOSE_ARN\"" \
    >> /tmp/lab102-env.sh
else
  echo "⚠️  Could not create Firehose"
fi

# Step 4: Enable WAF logging
echo ""
echo "[4] Enabling WAF logging to Firehose..."

# Wait for Firehose to be active
```

```bash
#!/bin/bash
# Lab 102 - Part 5: Automated IP Blocking
# Day 37 - AWS Security Specialist Study Plan

source /tmp/lab102-env.sh

echo "================================================"
echo "  Part 5: Automated IP Blocking Pipeline"
echo "================================================"

# Step 1: Create Lambda for automated blocking
echo ""
echo "[1] Creating automated IP blocking Lambda..."

mkdir -p /tmp/auto-block-lambda
cat > /tmp/auto-block-lambda/lambda_function.py << PYEOF
"""
Automated WAF IP blocking Lambda
Triggered by CloudWatch alarm → SNS → Lambda
Adds offending IP to WAF blocklist automatically
"""
import boto3
import json
import logging
import re

logger = logging.getLogger()
logger.setLevel(logging.INFO)

WAF_SCOPE = 'REGIONAL'
IP_SET_NAME = 'Lab102-BlockedIPs'

def lambda_handler(event, context):
    """
    Handles SNS notification from CloudWatch alarm
    Extracts IP from alarm and adds to WAF IP set
    """
    waf = boto3.client('wafv2')

    logger.info(f"Event received: {json.dumps(event)}")

    # Parse SNS message
    for record in event.get('Records', []):
        if record.get('EventSource') == 'aws:sns':
            message = json.loads(
                record['Sns']['Message']
            )
            logger.info(f"Alarm: {message}")

    # In production: extract IP from alarm dimensions
    # For demo: show the pattern
    offending_ip = extract_ip_from_event(event)

    if offending_ip:
        block_ip(waf, offending_ip)
    else:
        logger.info("No IP to block in this event")

    return {'statusCode': 200}

def extract_ip_from_event(event):
    """Extract IP address from CloudWatch alarm event"""
    try:
        # Parse CloudWatch alarm message
        message = json.loads(
            event['Records'][0]['Sns']['Message']
        )

        # Extract from alarm description or dimensions
        description = message.get(
            'AlarmDescription', ''
        )

        # Simple IP extraction pattern
        ip_pattern = r'\b(?:\d{1,3}\.){3}\d{1,3}\b'
        ips = re.findall(ip_pattern, description)

        return f"{ips[0]}/32" if ips else None
    except Exception as e:
        logger.error(f"Error extracting IP: {e}")
        return None

def block_ip(waf, ip_cidr):
    """Add IP to WAF IP set"""
    try:
        # Get current IP set
        ip_sets = waf.list_ip_sets(Scope=WAF_SCOPE)
        ip_set = next(
            (s for s in ip_sets['IPSets']
             if s['Name'] == IP_SET_NAME),
            None
        )

        if not ip_set:
            logger.error(f"IP set {IP_SET_NAME} not found")
            return

        # Get current addresses and lock token
        current = waf.get_ip_set(
            Name=IP_SET_NAME,
            Scope=WAF_SCOPE,
            Id=ip_set['Id']
        )

        addresses = current['IPSet']['Addresses']
        lock_token = current['LockToken']

        # Add new IP if not already blocked
        if ip_cidr not in addresses:
            addresses.append(ip_cidr)

            waf.update_ip_set(
                Name=IP_SET_NAME,
                Scope=WAF_SCOPE,
                Id=ip_set['Id'],
                Addresses=addresses,
                LockToken=lock_token
            )
            logger.info(f"✅ Blocked IP: {ip_cidr}")
        else:
            logger.info(f"IP {ip_cidr} already blocked")

    except Exception as e:
        logger.error(f"Error blocking IP: {e}")
        raise
PYEOF

# Package Lambda
cd /tmp/auto-block-lambda
zip -r /tmp/auto-block-lambda.zip \
  lambda_function.py 2>/dev/null && \
  echo "✅ Lambda package created"

# Create Lambda execution role
cat > /tmp/autoblock-trust.json << 'EOF'
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Effect": "Allow",
      "Principal": {
        "Service": "lambda.amazonaws.com"
      },
      "Action": "sts:AssumeRole"
    }
  ]
}
EOF

AUTOBLOCK_ROLE_ARN=$(aws iam create-role \
  --role-name "Lab102-AutoBlockRole" \
  --assume-role-policy-document \
    file:///tmp/autoblock-trust.json \
  --query 'Role.Arn' \
  --output text 2>/dev/null)

echo "Auto-block role: $AUTOBLOCK_ROLE_ARN"
echo "export AUTOBLOCK_ROLE_ARN=\"$AUTOBLOCK_ROLE_ARN\"" \
  >> /tmp/lab102-env.sh

# Attach WAF update policy
cat > /tmp/autoblock-policy.json << EOF
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Effect": "Allow",
      "Action": [
        "wafv2:GetIPSet",
        "wafv2:UpdateIPSet",
        "wafv2:ListIPSets"
      ],
      "Resource": "$BLOCKED_IPS_ARN"
    },
    {
      "Effect": "Allow",
      "Action": [
        "wafv2:ListIPSets"
      ],
      "Resource": "*"
    },
    {
      "Effect": "Allow",
      "Action": [
        "logs:CreateLogGroup",
        "logs:CreateLogStream",
        "logs:PutLogEvents"
      ],
      "Resource": "*"
    }
  ]
}
EOF

aws iam put-role-policy \
  --role-name "Lab102-AutoBlockRole" \
  --policy-name "Lab102-AutoBlockPolicy" \
  --policy-document \
    file:///tmp/autoblock-policy.json \
  2>/dev/null && \
  echo "✅ Auto-block policy attached"

sleep 10  # Wait for IAM propagation

# Create Lambda function
AUTOBLOCK_LAMBDA_ARN=$(aws lambda create-function \
  --function-name "Lab102-AutoIPBlock" \
  --runtime python3.11 \
  --handler lambda_function.lambda_handler \
  --role $AUTOBLOCK_ROLE_ARN \

    Variables="{WAF_IP_SET_ID=$BLOCKED_IPS_ID}" \

  --output text 2>/dev/null)

if [ ! -z "$AUTOBLOCK_LAMBDA_ARN" ]; then
  echo "✅ Auto-block Lambda created"
  echo "   ARN: $AUTOBLOCK_LAMBDA_ARN"
  echo "export AUTOBLOCK_LAMBDA_ARN=\"$AUTOBLOCK_LAMBDA_ARN\"" \
    >> /tmp/lab102-env.sh
fi

# Step 2: Create CloudWatch alarm for high block rate
echo ""
echo "[2] Creating CloudWatch alarm for high block rate..."
aws cloudwatch put-metric-alarm \

  echo "⚠️  Could not create alarm"

# Step 3: Show full automated blocking architecture
echo ""
echo "[3] Automated blocking architecture summary..."
cat << 'EOF'
```

```bash
  --scope CLOUDFRONT \
  --region us-east-1 \
  --query \
    'WebACLs[?Name==`Lab102-CloudFrontWebACL`].Id' \
  --output text 2>/dev/null)

if [ ! -z "$CF_WEB_ACL_ID" ] && \
   [ "$CF_WEB_ACL_ID" != "None" ]; then
  aws wafv2 delete-web-acl \
    --name "Lab102-CloudFrontWebACL" \
    --scope CLOUDFRONT \
    --region us-east-1 \
    --id $CF_WEB_ACL_ID \
    --lock-token $(aws wafv2 get-web-acl \
      --name "Lab102-CloudFrontWebACL" \
      --scope CLOUDFRONT \
      --region us-east-1 \
      --id $CF_WEB_ACL_ID \
      --query 'LockToken' \
      --output text 2>/dev/null) \
    2>/dev/null && \
    echo "✅ CloudFront Web ACL deleted"
fi

# Step 4: Delete rule group
echo ""
echo "[4] Deleting custom rule group..."
aws wafv2 delete-rule-group \
  --name "Lab102-CustomRuleGroup" \
  --scope REGIONAL \
  --id $RULE_GROUP_ID \
  --lock-token $(aws wafv2 get-rule-group \
    --name "Lab102-CustomRuleGroup" \
    --scope REGIONAL \
    --id $RULE_GROUP_ID \
    --query 'LockToken' \
    --output text 2>/dev/null) \
  2>/dev/null && \
  echo "✅ Rule group deleted"

echo "[5] Deleting IP sets..."
for IP_SET_NAME in \
  "Lab102-BlockedIPs" \
  "Lab102-CorporateIPs"; do

    --scope REGIONAL \

      "IPSets[?Name=='$IP_SET_NAME'].Id" \
    --output text 2>/dev/null)

  if [ ! -z "$IP_SET_ID" ] && \
     [ "$IP_SET_ID" != "None" ]; then

      --name $IP_SET_NAME \
      --scope REGIONAL \
      --id $IP_SET_ID \
      --lock-token $(aws wafv2 get-ip-set \
        --name $IP_SET_NAME \
        --scope REGIONAL \
        --id $IP_SET_ID \
        --query 'LockToken' \
        --output text 2>/dev/null) \
      2>/dev/null && \
      echo "✅ Deleted IP set: $IP_SET_NAME"
  fi
done

# Step 6: Delete regex pattern set
echo ""
echo "[6] Deleting regex pattern set..."
aws wafv2 delete-regex-pattern-set \
  --name "Lab102-SQLiPatterns" \
  --scope REGIONAL \
  --id $REGEX_ID \
  --lock-token $(aws wafv2 get-regex-pattern-set \
    --name "Lab102-SQLiPatterns" \
    --scope REGIONAL \
    --id $REGEX_ID \
    --query 'LockToken' \
    --output text 2>/dev/null) \
  2>/dev/null && \
  echo "✅ Regex pattern set deleted"

# Step 7: Delete Lambda functions

aws lambda delete-function \

# Step 8: Delete IAM roles
echo ""
echo "[8] Cleaning up IAM roles..."
for ROLE in \
  "Lab102-FirehoseRole" \
  "Lab102-AutoBlockRole"; do
  # Detach inline policies
  aws iam delete-role-policy \

  aws iam delete-role \
    --role-name $ROLE 2>/dev/null && \

done

# Step 9: Delete Kinesis Firehose
echo ""
echo "[9] Deleting Kinesis Firehose..."
aws firehose delete-delivery-stream \
  --delivery-stream-name "aws-waf-logs-lab102" \

# Step 10: Delete S3 bucket
echo ""
echo "[10] Deleting WAF log S3 bucket..."
aws s3 rm s3://$WAF_LOG_BUCKET \
  --recursive 2>/dev/null
aws s3 rb s3://$WAF_LOG_BUCKET \

# Step 11: Delete EventBridge rules
echo ""
echo "[11] Deleting EventBridge rules..."
aws events delete-rule \
  --name "Lab102-DDoSDetection" \
  --force 2>/dev/null && \
  echo "✅ DDoS detection rule deleted"

# Step 12: Delete CloudWatch resources
echo ""
echo "[12] Deleting CloudWatch resources..."
aws cloudwatch delete-alarms \
  --alarm-names "Lab102-WAFHighBlockRate" \
  2>/dev/null && echo "✅ CloudWatch alarm deleted"

aws cloudwatch delete-dashboards \
  --dashboard-names "Lab102-WAF-Dashboard" \
  2>/dev/null && echo "✅ CloudWatch dashboard deleted"

aws logs delete-log-group \
  --log-group-name "aws-waf-logs/lab102" \
  2>/dev/null && echo "✅ Log group deleted"

# Step 13: Cleanup temp files
echo ""
echo "[13] Cleaning up temp files..."
rm -f /tmp/lab102-env.sh \
      /tmp/firehose-trust.json \

rm -rf /tmp/auto-block-lambda/
```
