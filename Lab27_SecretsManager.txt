## Part 1 — Create & Manage Secrets

```bash
#!/bin/bash
# Lab 100 - Part 1: Create & Manage Secrets
# Day 35 - AWS Security Specialist Study Plan

source /tmp/lab100-env.sh

echo "================================================"
echo "  Part 1: Create & Manage Secrets"
echo "================================================"

# Step 1: Create KMS CMK for secrets encryption
echo ""
echo "[1] Creating dedicated KMS CMK for secrets..."
SECRET_KEY_ID=$(aws kms create-key \
  --description "Lab100 CMK for Secrets Manager" \
  --key-usage ENCRYPT_DECRYPT \
  --key-spec SYMMETRIC_DEFAULT \
  --tags TagKey=Lab,TagValue=Lab100 \
  --query 'KeyMetadata.KeyId' \
  --output text 2>/dev/null)

if [ ! -z "$SECRET_KEY_ID" ]; then
  echo "✅ KMS CMK created: $SECRET_KEY_ID"
  aws kms create-alias \
    --alias-name "alias/lab100-secrets-key" \
    --target-key-id $SECRET_KEY_ID 2>/dev/null && \
    echo "✅ Alias created: alias/lab100-secrets-key"
  echo "export SECRET_KEY_ID=\"$SECRET_KEY_ID\"" \
    >> /tmp/lab100-env.sh
fi

# Step 2: Create RDS-style database credential secret
echo ""
echo "[2] Creating database credential secret..."
DB_SECRET_ARN=$(aws secretsmanager create-secret \
  --name "lab100/production/db-credentials" \
  --description "Lab100 RDS MySQL credentials" \
  --kms-key-id "alias/lab100-secrets-key" \
  --secret-string '{
    "username": "dbadmin",
    "password": "InitialP@ssword123!",
    "engine": "mysql",
    "host": "lab100db.cluster-xxxx.ap-south-1.rds.amazonaws.com",
    "port": 3306,
    "dbname": "appdb"
  }' \
  --tags Key=Lab,Value=Lab100 \
         Key=Environment,Value=Production \
         Key=Type,Value=DatabaseCredential \
  --query 'ARN' \
  --output text 2>/dev/null)

if [ ! -z "$DB_SECRET_ARN" ]; then
  echo "✅ Database secret created"
  echo "   ARN: $DB_SECRET_ARN"
  echo "export DB_SECRET_ARN=\"$DB_SECRET_ARN\"" \
    >> /tmp/lab100-env.sh
else
  echo "⚠️  Could not create secret"
fi

# Step 3: Create API key secret
echo ""
echo "[3] Creating API key secret..."
API_SECRET_ARN=$(aws secretsmanager create-secret \
  --name "lab100/production/payment-api-key" \
  --description "Lab100 Payment gateway API key" \
  --kms-key-id "alias/lab100-secrets-key" \
  --secret-string '{
    "api_key": "pk_live_lab100_test_key_xxxx",
    "api_secret": "sk_live_lab100_test_secret_xxxx",
    "endpoint": "https://api.payment-gateway.com/v2",
    "environment": "production"

  --tags Key=Lab,Value=Lab100 \
         Key=Type,Value=APIKey \
  --query 'ARN' \
  --output text 2>/dev/null)

if [ ! -z "$API_SECRET_ARN" ]; then
  echo "✅ API key secret created"
  echo "   ARN: $API_SECRET_ARN"
  echo "export API_SECRET_ARN=\"$API_SECRET_ARN\"" \
    >> /tmp/lab100-env.sh
fi

# Step 4: Retrieve secret value
echo ""
echo "[4] Retrieving database secret value..."
aws secretsmanager get-secret-value \
  --secret-id "lab100/production/db-credentials" \
  --query '{
    Name:Name,
    VersionId:VersionId,
    SecretString:SecretString,
    VersionStages:VersionStages,
    CreatedDate:CreatedDate
  }' \
  --output json 2>/dev/null | python3 -m json.tool

# Step 5: List secret versions
echo ""
echo "[5] Listing secret versions and staging labels..."
aws secretsmanager list-secret-version-ids \
  --secret-id "lab100/production/db-credentials" \
  --query 'Versions[*].{
    VersionId:VersionId,
    StagingLabels:VersionStages,
    CreatedDate:CreatedDate

    "port": 3306,
    "dbname": "appdb"

    VersionId:VersionId,
```

```bash
{

# Step 3: Create rotation Lambda function
echo ""
echo "[3] Creating rotation Lambda function..."

# Create Lambda deployment package
mkdir -p /tmp/rotation-lambda
cat > /tmp/rotation-lambda/lambda_function.py << 'LAMBDAEOF'
"""
Custom rotation Lambda for Lab 100
Implements the 4-step Secrets Manager rotation protocol
This is a simulation - in production connect to actual DB
"""
import boto3
import json
import logging
import os
import string
import random

logger = logging.getLogger()
logger.setLevel(logging.INFO)

def lambda_handler(event, context):
    """Main rotation handler - routes to correct step"""
    arn = event['SecretId']
    token = event['ClientRequestToken']
    step = event['Step']

    # Get Secrets Manager client
    service_client = boto3.client('secretsmanager')

    # Ensure secret exists and token is valid
    metadata = service_client.describe_secret(SecretId=arn)

    if not metadata['RotationEnabled']:
        raise ValueError(f"Rotation not enabled for {arn}")

    versions = metadata.get('VersionIdsToStages')
    if token not in versions:
        raise ValueError(f"Token {token} not in secret versions")

    if "AWSCURRENT" in versions[token]:
        logger.info("Token is already AWSCURRENT - no rotation needed")
        return

    if "AWSPENDING" not in versions[token]:
        raise ValueError(f"Token {token} not set as AWSPENDING")

    # Route to appropriate step
    if step == "createSecret":
        create_secret(service_client, arn, token)
    elif step == "setSecret":
        set_secret(service_client, arn, token)
    elif step == "testSecret":
        test_secret(service_client, arn, token)
    elif step == "finishSecret":
        finish_secret(service_client, arn, token)
    else:
        raise ValueError(f"Invalid step: {step}")

def create_secret(client, arn, token):
    """Step 1: Generate new password and store as AWSPENDING"""
    logger.info(f"createSecret: {arn}")

    try:
        # Check if AWSPENDING already exists
        client.get_secret_value(
            SecretId=arn,
            VersionStage="AWSPENDING",
            VersionId=token
        )
        logger.info("AWSPENDING already exists - skipping creation")
        return
    except client.exceptions.ResourceNotFoundException:
        pass

    # Get current secret
    current = json.loads(
        client.get_secret_value(
            SecretId=arn,
            VersionStage="AWSCURRENT"
        )['SecretString']
    )

    # Generate new password
    new_password = generate_password()

    # Create new version with AWSPENDING
    current['password'] = new_password
    client.put_secret_value(
        SecretId=arn,
        ClientRequestToken=token,
        SecretString=json.dumps(current),
        VersionStages=['AWSPENDING']
    )
    logger.info("✅ New password created as AWSPENDING")

def set_secret(client, arn, token):
    """Step 2 & 3: Apply new password to database"""
    logger.info(f"setSecret: {arn}")

    # Get AWSPENDING credentials
    pending = json.loads(
        client.get_secret_value(
            SecretId=arn,
            VersionStage="AWSPENDING",
            VersionId=token
        )['SecretString']
    )

    # In production: connect to DB and change password
    # db_connection = connect_to_db(pending)
    # db_connection.execute(f"ALTER USER '{pending['username']}'
    #   IDENTIFIED BY '{pending['password']}'")

    logger.info(f"✅ Password updated for user: {pending['username']}")
    logger.info("(Simulated - in production would update actual DB)")

def test_secret(client, arn, token):
    """Step 2 (continued): Test new credentials work"""
    logger.info(f"testSecret: {arn}")

    pending = json.loads(
        client.get_secret_value(
            SecretId=arn,
            VersionStage="AWSPENDING",
            VersionId=token
        )['SecretString']
    )

    # In production: verify new credentials actually work
    # try:
    #     conn = connect_to_db(pending)
    #     conn.execute("SELECT 1")
    #     logger.info("✅ New credentials verified")
    # except Exception as e:
    #     raise ValueError(f"New credentials invalid: {e}")

    logger.info("✅ Credentials tested (simulated)")

def finish_secret(client, arn, token):
    """Step 4: Promote AWSPENDING to AWSCURRENT"""
    logger.info(f"finishSecret: {arn}")

    # Get current version ID
    metadata = client.describe_secret(SecretId=arn)
    current_version = None
    for version, stages in \
        metadata['VersionIdsToStages'].items():
        if 'AWSCURRENT' in stages:
            if version == token:
                logger.info("Already AWSCURRENT - no action needed")
                return
            current_version = version
            break

    # Promote AWSPENDING to AWSCURRENT
    client.update_secret_version_stage(
        SecretId=arn,
        VersionStage='AWSCURRENT',
        MoveToVersionId=token,
        RemoveFromVersionId=current_version
    )
    logger.info("✅ AWSPENDING promoted to AWSCURRENT")

def generate_password(length=32):
    """Generate a strong random password"""
    chars = (
        string.ascii_letters +
        string.digits +
        "!@#$%^&*"
    )
    password = (
        random.choice(string.ascii_uppercase) +
        random.choice(string.ascii_lowercase) +
        random.choice(string.digits) +
        random.choice("!@#$%^&*") +
        ''.join(random.choices(chars, k=length-4))
    )
    return ''.join(random.sample(password, len(password)))
LAMBDAEOF

# Package Lambda
cd /tmp/rotation-lambda
zip -r /tmp/rotation-lambda.zip lambda_function.py \
  2>/dev/null && echo "✅ Lambda package created"

# Wait for IAM role to propagate
echo "   Waiting 10s for IAM role to propagate..."
sleep 10

# Create Lambda function
ROTATION_LAMBDA_ARN=$(aws lambda create-function \
  --function-name "Lab100-SecretsRotation" \
  --runtime python3.11 \
  --handler lambda_function.lambda_handler \
  --role $ROTATION_ROLE_ARN \
  --zip-file fileb:///tmp/rotation-lambda.zip \

  --query 'FunctionArn' \

  echo "✅ Rotation Lambda created"
  echo "   ARN: $ROTATION_LAMBDA_ARN"
  echo "export ROTATION_LAMBDA_ARN=\"$ROTATION_LAMBDA_ARN\"" \
    >> /tmp/lab100-env.sh
else
  echo "⚠️  Could not create Lambda function"
fi

# Step 4: Grant Secrets Manager permission to invoke Lambda
echo ""
echo "[4] Granting Secrets Manager permission to invoke Lambda..."
aws lambda add-permission \
  --function-name "Lab100-SecretsRotation" \
  --statement-id "SecretsManagerInvoke" \
  --action lambda:InvokeFunction \
  --principal secretsmanager.amazonaws.com \
  --source-account $ACCOUNT_ID 2>/dev/null && \
  echo "✅ Lambda invoke permission granted" || \
  echo "⚠️  Could not add Lambda permission"

# Step 5: Enable rotation on the secret
echo ""
echo "[5] Enabling automatic rotation on DB secret..."
aws secretsmanager rotate-secret \
  --secret-id "lab100/production/db-credentials" \
  --rotation-lambda-arn $ROTATION_LAMBDA_ARN \
  --rotation-rules '{
    "AutomaticallyAfterDays": 30
  }' 2>/dev/null && \
  echo "✅ Rotation enabled (every 30 days)" || \
  echo "⚠️  Could not enable rotation"

# Step 6: Verify rotation configuration
echo ""
echo "[6] Verifying rotation configuration..."
aws secretsmanager describe-secret \
  --secret-id "lab100/production/db-credentials" \
  --query '{
    Name:Name,
    RotationEnabled:RotationEnabled,
    RotationLambdaARN:RotationLambdaARN,
    RotationRules:RotationRules,
    LastRotatedDate:LastRotatedDate
  }' \
  --output table 2>/dev/null

# Step 7: Trigger immediate rotation
echo ""
echo "[7] Triggering immediate rotation test..."
aws secretsmanager rotate-secret \
  --secret-id "lab100/production/db-credentials" 2>/dev/null && \
  echo "✅ Rotation triggered manually" || \
  echo "⚠️  Could not trigger rotation"

# Step 8: Check rotation status
echo ""
echo "[8] Checking rotation status after trigger..."
sleep 5
aws secretsmanager describe-secret \
  --secret-id "lab100/production/db-credentials" \
  --query '{
    RotationEnabled:RotationEnabled,
    LastRotatedDate:LastRotatedDate,
    LastChangedDate:LastChangedDate
  }' \
  --output table 2>/dev/null

echo ""
echo "================================================"
echo "Part 2 Complete!"
echo "Rotation configured:"
echo "  - Custom Lambda with 4-step protocol"
echo "  - Automatic every 30 days"
echo "  - Manual trigger tested"
echo "================================================"
```

## Part 3 — Resource Policy & Cross-Account Setup

```bash
#!/bin/bash
# Lab 100 - Part 3: Resource Policy Setup
# Day 35 - AWS Security Specialist Study Plan

source /tmp/lab100-env.sh

echo "================================================"
echo "  Part 3: Resource Policy & Cross-Account Setup"
echo "================================================"

# Step 1: Create resource policy on secret
echo ""
echo "[1] Creating resource policy on DB secret..."
cat > /tmp/secret-resource-policy.json << EOF
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Sid": "AllowCurrentAccountAccess",
      "Effect": "Allow",
      "Principal": {
        "AWS": "arn:aws:iam::${ACCOUNT_ID}:root"
      },
      "Action": [
        "secretsmanager:GetSecretValue",
        "secretsmanager:DescribeSecret"
      ],
      "Resource": "*"
    },
    {
      "Sid": "AllowCrossAccountAccess",
      "Effect": "Allow",
      "Principal": {
        "AWS": "arn:aws:iam::ACCOUNT_B_ID:role/AppRole"
      },
      "Action": [
        "secretsmanager:GetSecretValue",
        "secretsmanager:DescribeSecret"
      ],
      "Resource": "*",
      "Condition": {
        "StringEquals": {
          "secretsmanager:VersionStage": "AWSCURRENT"
        }
      }
    },
    {
      "Sid": "DenyInsecureTransport",
      "Effect": "Deny",
      "Principal": "*",
      "Action": "secretsmanager:*",
      "Resource": "*",
      "Condition": {
        "Bool": {
          "aws:SecureTransport": "false"
        }
      }
    }
  ]
}
EOF

aws secretsmanager put-resource-policy \
  --secret-id "lab100/production/db-credentials" \
  --resource-policy \
    file:///tmp/secret-resource-policy.json \
  --block-public-policy 2>/dev/null && \
  echo "✅ Resource policy applied" || \
  echo "⚠️  Could not apply resource policy"

# Step 2: View the resource policy
echo ""
echo "[2] Viewing applied resource policy..."
aws secretsmanager get-resource-policy \
  --secret-id "lab100/production/db-credentials" \
  --query 'ResourcePolicy' \
  --output text 2>/dev/null | python3 -m json.tool

# Step 3: Update KMS key policy for cross-account
echo ""
echo "[3] Updating KMS key policy for cross-account access..."
cat > /tmp/cross-account-key-policy.json << EOF
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Sid": "EnableIAMUserPermissions",
      "Effect": "Allow",
      "Principal": {
        "AWS": "arn:aws:iam::${ACCOUNT_ID}:root"
      },
      "Action": "kms:*",
      "Resource": "*"
    },
    {
      "Sid": "AllowSecretsManagerToUseKey",
      "Effect": "Allow",
      "Principal": {
        "Service": "secretsmanager.amazonaws.com"
      },
      "Action": [
        "kms:GenerateDataKey",
        "kms:Decrypt"
      ],
      "Resource": "*"
    },
    {
      "Sid": "AllowCrossAccountDecrypt",
      "Effect": "Allow",
      "Principal": {
        "AWS": "arn:aws:iam::ACCOUNT_B_ID:role/AppRole"
      },
      "Action": [
        "kms:Decrypt",
        "kms:DescribeKey"
      ],
      "Resource": "*"
    }
  ]
}
EOF

aws kms put-key-policy \
  --key-id $SECRET_KEY_ID \
  --policy-name default \
  --policy \
    file:///tmp/cross-account-key-policy.json \
  2>/dev/null && \
  echo "✅ KMS key policy updated for cross-account" || \
  echo "⚠️  Could not update KMS key policy"

# Step 4: Show Account B side requirements
echo ""
echo "[4] Account B side requirements reference..."
cat << 'EOF'
# In Account B, the AppRole needs this IAM policy:
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Effect": "Allow",
      "Action": [
        "secretsmanager:GetSecretValue",
        "secretsmanager:DescribeSecret"
      ],
      "Resource": "arn:aws:secretsmanager:<region>:<AccountA>:secret:lab100/production/db-credentials-*"
    },
    {
      "Effect": "Allow",
      "Action": [
        "kms:Decrypt",
        "kms:DescribeKey"
      ],
      "Resource": "arn:aws:kms:<region>:<AccountA>:key/<SECRET_KEY_ID>"
    }
  ]
}

# Three-part requirement confirmed:
# 1. Secret resource policy (Account A) ✅ Done
# 2. KMS key policy (Account A)         ✅ Done
# 3. IAM policy (Account B)             → Account B team applies
EOF

# Step 5: Validate resource policy
echo ""
echo "[5] Validating resource policy..."
aws secretsmanager validate-resource-policy \
  --resource-policy \
    file:///tmp/secret-resource-policy.json \
  2>/dev/null && \
  echo "✅ Resource policy is valid" || \
  echo "⚠️  Resource policy validation failed"

echo ""
echo "================================================"
echo "Part 3 Complete!"
echo "Cross-account setup configured:"
echo "  1. Secret resource policy with cross-acct access"
echo "  2. KMS key policy allowing cross-acct decrypt"
echo "  3. Account B IAM policy template provided"
echo "  4. Deny insecure transport added"
echo "================================================"
```

```bash
#!/bin/bash
# Lab 100 - Part 4: Parameter Store Hierarchy
# Day 35 - AWS Security Specialist Study Plan

source /tmp/lab100-env.sh

echo "================================================"
```

```bash
import boto3
import json
import time

def get_secret_with_caching():
    """
    Demonstrates secret retrieval with in-memory caching
    In production use AWS Secrets Manager caching client
    """
    client = boto3.client('secretsmanager')

    # Simple cache implementation
    cache = {}
    cache_ttl = 300  # 5 minutes

    def get_secret(secret_name):
        now = time.time()

        # Check cache
        if secret_name in cache:
            cached_value, cached_time = cache[secret_name]
            if now - cached_time < cache_ttl:
                print(f"  Cache HIT for: {secret_name}")
                return cached_value

        # Cache miss — fetch from Secrets Manager
        print(f"  Cache MISS — fetching: {secret_name}")
        response = client.get_secret_value(
            SecretId=secret_name
        )
        secret = json.loads(response['SecretString'])

        # Cache the result
        cache[secret_name] = (secret, now)
        return secret

    # Simulate multiple calls
    print("\n=== Secrets Manager Caching Demo ===")
    for i in range(3):
        print(f"\nCall {i+1}:")
        secret = get_secret('lab100/production/db-credentials')
        print(f"  Username: {secret.get('username')}")
        print(f"  Host: {secret.get('host')}")
        print(f"  DB: {secret.get('dbname')}")

    print("\n✅ Only 1 API call made — 2 served from cache")

import urllib.request

def get_secret_via_extension(secret_name):
    """
    Use Lambda extension for fast local secret retrieval
    Extension caches secrets — reduces latency and cost
    """
    headers = {
```
