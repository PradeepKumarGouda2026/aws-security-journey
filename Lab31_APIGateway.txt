## Part 1 — Create REST API with Resources

```bash
#!/bin/bash
# Lab 104 - Part 1: Create REST API
# Day 39 - AWS Security Specialist Study Plan

source /tmp/lab104-env.sh

echo "================================================"
echo "  Part 1: Create REST API with Resources"
echo "================================================"

# Step 1: Create REST API
echo ""
echo "[1] Creating REST API..."
API_ID=$(aws apigateway create-rest-api \
  --name "Lab104-SecureAPI" \
  --description "Lab104 API Gateway Security Demo" \
  --endpoint-configuration \
    types=REGIONAL \
  --api-key-source HEADER \
  --tags Lab=Lab104 Environment=Production \
  --query 'id' \
  --output text 2>/dev/null)

if [ ! -z "$API_ID" ]; then
  echo "✅ REST API created"
  echo "   API ID: $API_ID"
  echo "export API_ID=\"$API_ID\"" \

fi

# Step 2: Get root resource ID

  get-resources \

  --output text 2>/dev/null)

echo "   Root Resource ID: $ROOT_RESOURCE_ID"
echo "export ROOT_RESOURCE_ID=\"$ROOT_RESOURCE_ID\"" \
  >> /tmp/lab104-env.sh

# Step 3: Create /users resource
echo ""
echo "[2] Creating API resources..."
USERS_RESOURCE_ID=$(aws apigateway \
  create-resource \
  --rest-api-id $API_ID \
  --parent-id $ROOT_RESOURCE_ID \
  --path-part "users" \
  --query 'id' \
  --output text 2>/dev/null)

echo "✅ /users resource created: $USERS_RESOURCE_ID"
echo "export USERS_RESOURCE_ID=\"$USERS_RESOURCE_ID\"" \
  >> /tmp/lab104-env.sh

# Create /admin resource
ADMIN_RESOURCE_ID=$(aws apigateway \
  create-resource \
  --rest-api-id $API_ID \
  --parent-id $ROOT_RESOURCE_ID \
  --path-part "admin" \
  --query 'id' \
  --output text 2>/dev/null)

echo "✅ /admin resource created: $ADMIN_RESOURCE_ID"
echo "export ADMIN_RESOURCE_ID=\"$ADMIN_RESOURCE_ID\"" \
  >> /tmp/lab104-env.sh

# Create /public resource
PUBLIC_RESOURCE_ID=$(aws apigateway \
  create-resource \
  --rest-api-id $API_ID \
  --parent-id $ROOT_RESOURCE_ID \
  --path-part "public" \
  --query 'id' \
  --output text 2>/dev/null)

echo "✅ /public resource created: $PUBLIC_RESOURCE_ID"
echo "export PUBLIC_RESOURCE_ID=\"$PUBLIC_RESOURCE_ID\"" \
  >> /tmp/lab104-env.sh

# Create /login resource for rate limiting demo
LOGIN_RESOURCE_ID=$(aws apigateway \
  create-resource \
  --rest-api-id $API_ID \
  --parent-id $ROOT_RESOURCE_ID \
  --path-part "login" \
  --query 'id' \
  --output text 2>/dev/null)

echo "✅ /login resource created: $LOGIN_RESOURCE_ID"
echo "export LOGIN_RESOURCE_ID=\"$LOGIN_RESOURCE_ID\"" \
  >> /tmp/lab104-env.sh

# Step 4: Create backend Lambda function
echo ""
echo "[3] Creating backend Lambda function..."

mkdir -p /tmp/lab104-lambda

cat > /tmp/lab104-lambda/backend.py << 'PYEOF'
"""
Lab 104 - Backend Lambda function
Simulates API responses with user context
"""
import json
import logging

logger = logging.getLogger()
logger.setLevel(logging.INFO)

def lambda_handler(event, context):
    logger.info(f"Event: {json.dumps(event)}")

    # Extract authorizer context
    request_context = event.get(
        'requestContext', {}
    )
    authorizer = request_context.get(
        'authorizer', {}
    )

    # Get path
    path = event.get('path', '/')
    method = event.get('httpMethod', 'GET')

    # Build response based on path
    if '/users' in path:
        body = {
            'message': 'Users endpoint',
            'authenticatedUser': authorizer.get(
                'principalId', 'unknown'
            ),
            'userRole': authorizer.get(
                'role', 'unknown'
            ),
            'userId': authorizer.get(
                'userId', 'unknown'
            )
        }
    elif '/admin' in path:
        body = {
            'message': 'Admin endpoint',
            'adminUser': authorizer.get(
                'principalId', 'unknown'
            ),
            'permissions': authorizer.get(
                'permissions', 'none'
            )
        }
    elif '/public' in path:
        body = {
            'message': 'Public endpoint - no auth needed',
            'path': path
        }
    elif '/login' in path:
        body = {
            'message': 'Login endpoint',
            'rateLimited': False
        }
    else:
        body = {
            'message': f'{method} {path}',
            'context': authorizer
        }

    return {
        'statusCode': 200,
        'headers': {
            'Content-Type': 'application/json',
            'Access-Control-Allow-Origin':
                'https://app.lab104.com',
            'Access-Control-Allow-Methods':
                'GET,POST,OPTIONS',
            'Access-Control-Allow-Headers':
                'Authorization,Content-Type'
        },
        'body': json.dumps(body)
    }
PYEOF

# Create IAM role for backend Lambda
cat > /tmp/backend-trust.json << 'EOF'
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Effect": "Allow",
      "Principal": {
        "Service": "lambda.amazonaws.com"
      },
      "Action": "sts:AssumeRole"
    }
  ]
}
EOF

BACKEND_ROLE_ARN=$(aws iam create-role \
  --role-name "Lab104-BackendLambdaRole" \
  --assume-role-policy-document \
    file:///tmp/backend-trust.json \
  --query 'Role.Arn' \
  --output text 2>/dev/null)

aws iam attach-role-policy \
  --role-name "Lab104-BackendLambdaRole" \
  --policy-arn \
    "arn:aws:iam::aws:policy/service-role/AWSLambdaBasicExecutionRole" \
  2>/dev/null

echo "✅ Backend Lambda role created"
echo "export BACKEND_ROLE_ARN=\"$BACKEND_ROLE_ARN\"" \
  >> /tmp/lab104-env.sh

sleep 10  # Wait for IAM propagation

# Package and deploy backend Lambda
cd /tmp/lab104-lambda
zip backend.zip backend.py 2>/dev/null

BACKEND_LAMBDA_ARN=$(aws lambda create-function \
  --function-name "Lab104-BackendLambda" \
  --runtime python3.11 \
  --handler backend.lambda_handler \
  --role $BACKEND_ROLE_ARN \
  --zip-file fileb:///tmp/lab104-lambda/backend.zip \
  --description "Lab104 backend API handler" \
  --timeout 10 \
  --tags Lab=Lab104 \
  --query 'FunctionArn' \
  --output text 2>/dev/null)

echo "✅ Backend Lambda created: $BACKEND_LAMBDA_ARN"
echo "export BACKEND_LAMBDA_ARN=\"$BACKEND_LAMBDA_ARN\"" \
  >> /tmp/lab104-env.sh

# Step 5: Add Lambda permission for API Gateway
echo ""
echo "[4] Granting API Gateway permission to invoke Lambda..."
aws lambda add-permission \
  --function-name "Lab104-BackendLambda" \
  --statement-id "APIGatewayInvoke" \
  --action lambda:InvokeFunction \
  --principal apigateway.amazonaws.com \
  --source-arn \
    "arn:aws:execute-api:${REGION}:${ACCOUNT_ID}:${API_ID}/*" \
  2>/dev/null && \
  echo "✅ Lambda permission granted"

# Step 6: Create GET methods on resources
echo ""
echo "[5] Creating GET methods on resources..."

for RESOURCE_INFO in \
  "$USERS_RESOURCE_ID:NONE" \
  "$PUBLIC_RESOURCE_ID:NONE" \
  "$LOGIN_RESOURCE_ID:NONE"; do

  RESOURCE_ID=$(echo $RESOURCE_INFO | cut -d: -f1)
  AUTH_TYPE=$(echo $RESOURCE_INFO | cut -d: -f2)

  aws apigateway put-method \
    --rest-api-id $API_ID \
    --resource-id $RESOURCE_ID \
    --http-method GET \
    --authorization-type $AUTH_TYPE \
    --api-key-required false \
    2>/dev/null && \
    echo "✅ GET method created on resource: $RESOURCE_ID"

  # Add Lambda integration
  aws apigateway put-integration \
    --rest-api-id $API_ID \
    --resource-id $RESOURCE_ID \
    --http-method GET \
    --type AWS_PROXY \
    --integration-http-method POST \
    --uri "arn:aws:apigateway:${REGION}:lambda:path/2015-03-31/functions/${BACKEND_LAMBDA_ARN}/invocations" \
    2>/dev/null && \
    echo "   ✅ Lambda integration added"
done

echo ""
echo "================================================"
echo "Part 1 Complete!"
echo "REST API: $API_ID"
echo "Resources created:"
echo "  /users  (auth: to be configured)"
echo "  /admin  (auth: to be configured)"
echo "  /public (auth: NONE)"
echo "  /login  (auth: NONE + rate limit)"
echo "================================================"
```

## Part 2 — Configure Authorizers

```bash
#!/bin/bash
# Lab 104 - Part 2: Configure Authorizers
# Day 39 - AWS Security Specialist Study Plan

source /tmp/lab104-env.sh

echo "================================================"
echo "  Part 2: Configure Authorizers"
echo "================================================"

# Step 1: Create Lambda Authorizer function
echo ""
echo "[1] Creating Lambda Authorizer function..."

cat > /tmp/lab104-lambda/authorizer.py << 'PYEOF'
"""
Lab 104 - Lambda Authorizer (TOKEN type)
Validates custom bearer tokens
"""
import json
import logging
import re

logger = logging.getLogger()
logger.setLevel(logging.INFO)

# Simulated token database
# In production: validate against DB/cache/JWT library
VALID_TOKENS = {
    "admin-token-secret-123": {
        "userId": "user-001",
        "role": "admin",
        "permissions": "read,write,delete,admin"
    },
    "user-token-secret-456": {
        "userId": "user-002",
        "role": "user",
        "permissions": "read"
    },
    "premium-token-secret-789": {
        "userId": "user-003",
        "role": "premium",
        "permissions": "read,write"
    }
}

def lambda_handler(event, context):
    logger.info(f"Authorizer event: {json.dumps(event)}")

    token = event.get('authorizationToken', '')
    method_arn = event.get('methodArn', '')

    # Remove "Bearer " prefix if present
    if token.startswith('Bearer '):
        token = token[7:]

    # Validate token
    token_data = VALID_TOKENS.get(token)

    if token_data:
        logger.info(
            f"Valid token for user: {token_data['userId']}"
        )
        return generate_policy(
            principal_id=token_data['userId'],
            effect='Allow',
            resource=method_arn,
            context={
                'userId': token_data['userId'],
                'role': token_data['role'],
                'permissions': token_data['permissions']
            }
        )
    else:
        logger.warning(
            f"Invalid token received"
        )
        # Fail closed — raise exception for 401
        raise Exception('Unauthorized')

def generate_policy(
    principal_id, effect, resource, context=None
):
    """Generate IAM policy for API Gateway"""

    # Allow/deny all methods for simplicity
    # In production: be more specific about resources
    resource_parts = resource.split(':')
    region = resource_parts[3]
    account = resource_parts[4]
    api_gateway_arn = ':'.join(resource_parts[:6])
    stage_arn = api_gateway_arn.split('/')[0]

    policy = {
        'principalId': principal_id,
        'policyDocument': {
            'Version': '2012-10-17',
            'Statement': [
                {
                    'Action': 'execute-api:Invoke',
                    'Effect': effect,
                    'Resource': f"{stage_arn}/*"
                }
            ]
        }
    }

    if context:
        policy['context'] = context

    logger.info(f"Generated policy: {json.dumps(policy)}")
    return policy
PYEOF

# Package authorizer Lambda
cd /tmp/lab104-lambda
zip authorizer.zip authorizer.py 2>/dev/null

AUTHORIZER_LAMBDA_ARN=$(aws lambda create-function \
  --function-name "Lab104-LambdaAuthorizer" \
  --runtime python3.11 \
  --handler authorizer.lambda_handler \
  --role $BACKEND_ROLE_ARN \
  --zip-file \
    fileb:///tmp/lab104-lambda/authorizer.zip \
  --description "Lab104 Lambda TOKEN authorizer" \
  --timeout 5 \
  --tags Lab=Lab104 \
  --query 'FunctionArn' \
  --output text 2>/dev/null)

echo "✅ Lambda Authorizer created: $AUTHORIZER_LAMBDA_ARN"
echo "export AUTHORIZER_LAMBDA_ARN=\"$AUTHORIZER_LAMBDA_ARN\"" \
  >> /tmp/lab104-env.sh

# Grant API Gateway permission to invoke authorizer
aws lambda add-permission \
  --function-name "Lab104-LambdaAuthorizer" \
  --statement-id "APIGatewayInvokeAuthorizer" \
  --action lambda:InvokeFunction \
  --principal apigateway.amazonaws.com \
  --source-arn \
    "arn:aws:execute-api:${REGION}:${ACCOUNT_ID}:${API_ID}/*" \
  2>/dev/null && \
  echo "✅ Authorizer Lambda permission granted"

# Step 2: Create Lambda Authorizer in API Gateway
echo ""
echo "[2] Creating Lambda Authorizer in API Gateway..."
LAMBDA_AUTHORIZER_ID=$(aws apigateway \
  create-authorizer \
  --rest-api-id $API_ID \
  --name "Lab104-LambdaAuth" \
  --type TOKEN \
  --authorizer-uri \
    "arn:aws:apigateway:${REGION}:lambda:path/2015-03-31/functions/${AUTHORIZER_LAMBDA_ARN}/invocations" \
  --identity-source "method.request.header.Authorization" \
  --authorizer-result-ttl-in-seconds 300 \
  --query 'id' \
  --output text 2>/dev/null)

echo "✅ Lambda Authorizer created in API GW"
echo "   Authorizer ID: $LAMBDA_AUTHORIZER_ID"
echo "export LAMBDA_AUTHORIZER_ID=\"$LAMBDA_AUTHORIZER_ID\"" \
  >> /tmp/lab104-env.sh

# Step 3: Apply Lambda Authorizer to /users GET
echo ""
echo "[3] Applying Lambda Authorizer to /users endpoint..."
aws apigateway update-method \
  --rest-api-id $API_ID \
  --resource-id $USERS_RESOURCE_ID \
  --http-method GET \
  --patch-operations \
    op=replace,path=/authorizationType,\
value=CUSTOM \
    op=replace,path=/authorizerId,\
value=$LAMBDA_AUTHORIZER_ID \
  2>/dev/null && \
  echo "✅ Lambda Authorizer applied to /users GET"

# Step 4: Create IAM Authorizer on /admin
echo ""
echo "[4] Applying IAM Authorization to /admin endpoint..."
aws apigateway put-method \
  --rest-api-id $API_ID \
  --resource-id $ADMIN_RESOURCE_ID \
  --http-method GET \
  --authorization-type AWS_IAM \
  --api-key-required false \
  2>/dev/null && \
  echo "✅ IAM Authorization applied to /admin GET"

# Add Lambda integration to /admin
aws apigateway put-integration \
  --rest-api-id $API_ID \
  --resource-id $ADMIN_RESOURCE_ID \
  --http-method GET \
  --type AWS_PROXY \
  --integration-http-method POST \
  --uri "arn:aws:apigateway:${REGION}:lambda:path/2015-03-31/functions/${BACKEND_LAMBDA_ARN}/invocations" \
  2>/dev/null && \
  echo "✅ Lambda integration added to /admin"

# Step 5: List all authorizers
echo ""
echo "[5] Listing authorizers on API..."
aws apigateway get-authorizers \
  --rest-api-id $API_ID \
  --query 'items[*].{
    Name:name,
    Id:id,
    Type:type,
    TTL:authorizerResultTtlInSeconds
  }' \
  --output table 2>/dev/null

echo ""
echo "================================================"
echo "Part 2 Complete!"
echo "Authorizers configured:"
echo "  /users GET → Lambda Authorizer (TOKEN)"
echo "  /admin GET → IAM Authorization (SigV4)"
echo "  /public GET → NONE (open)"
echo "  /login GET → NONE + rate limit (next part)"
echo "================================================"
```

## Part 3 — Resource Policy, Throttling & Validation

```bash
#!/bin/bash
# Lab 104 - Part 3: Resource Policy, Throttling & Validation
# Day 39 - AWS Security Specialist Study Plan

source /tmp/lab104-env.sh

echo "================================================"
echo "  Part 3: Resource Policy, Throttling & Validation"
echo "================================================"

# Step 1: Create request model for validation
echo ""
echo "[1] Creating request validation model..."
MODEL_ID=$(aws apigateway create-model \
  --rest-api-id $API_ID \
  --name "Lab104LoginModel" \
  --content-type "application/json" \
  --schema '{
    "$schema": "http://json-schema.org/draft-04/schema#",
    "title": "LoginRequest",
    "type": "object",
    "properties": {
      "username": {
        "type": "string",
        "minLength": 3,
        "maxLength": 100
      },
      "password": {
        "type": "string",
        "minLength": 8
      },
      "mfaCode": {
        "type": "string",
        "pattern": "^[0-9]{6}$"
      }
    },
    "required": ["username", "password"]
  }' \
  --query 'id' \
  --output text 2>/dev/null)

echo "✅ Request model created: $MODEL_ID"
echo "export MODEL_ID=\"$MODEL_ID\"" \
  >> /tmp/lab104-env.sh

# Step 2: Create request validator
echo ""
echo "[2] Creating request validator..."
VALIDATOR_ID=$(aws apigateway \
  create-request-validator \
  --rest-api-id $API_ID \
  --name "Lab104-BodyValidator" \
  --validate-request-body \
  --no-validate-request-parameters \
  --query 'id' \
  --output text 2>/dev/null)

echo "✅ Request validator created: $VALIDATOR_ID"
echo "export VALIDATOR_ID=\"$VALIDATOR_ID\"" \
  >> /tmp/lab104-env.sh

# Step 3: Create POST method on /login with validation
echo ""
echo "[3] Creating POST /login with request validation..."
aws apigateway put-method \
  --rest-api-id $API_ID \
  --resource-id $LOGIN_RESOURCE_ID \
  --http-method POST \
  --authorization-type NONE \
  --request-validator-id $VALIDATOR_ID \
  --request-models \
    "application/json=$MODEL_ID" \
  2>/dev/null && \
  echo "✅ POST /login created with body validation"

# Add Lambda integration
aws apigateway put-integration \
  --rest-api-id $API_ID \
  --resource-id $LOGIN_RESOURCE_ID \
  --http-method POST \
  --type AWS_PROXY \
  --integration-http-method POST \
  --uri "arn:aws:apigateway:${REGION}:lambda:path/2015-03-31/functions/${BACKEND_LAMBDA_ARN}/invocations" \
  2>/dev/null && \
  echo "✅ Lambda integration added to POST /login"

# Step 4: Deploy API to stage
echo ""
echo "[4] Deploying API to production stage..."
DEPLOYMENT_ID=$(aws apigateway \
  create-deployment \
  --rest-api-id $API_ID \
  --stage-name prod \
  --stage-description "Lab104 Production Stage" \
  --description "Lab104 initial deployment" \
  --query 'id' \
  --output text 2>/dev/null)

echo "✅ API deployed to prod stage"
echo "   Deployment ID: $DEPLOYMENT_ID"
echo "export DEPLOYMENT_ID=\"$DEPLOYMENT_ID\"" \
  >> /tmp/lab104-env.sh

# Step 5: Configure stage-level throttling
echo ""
echo "[5] Configuring stage-level throttling..."
aws apigateway update-stage \
  --rest-api-id $API_ID \
  --stage-name prod \
  --patch-operations \
    op=replace,path=/defaultRouteSettings/throttlingRateLimit,\
value=1000 \
    op=replace,path=/defaultRouteSettings/throttlingBurstLimit,\
value=2000 \
  2>/dev/null || \
  aws apigateway update-stage \
    --rest-api-id $API_ID \
    --stage-name prod \
    --patch-operations \
      "op=replace,path=/*/*/throttling/rateLimit,value=1000" \
      "op=replace,path=/*/*/throttling/burstLimit,value=2000" \
    2>/dev/null && \
    echo "✅ Stage throttling configured (1000 RPS, 2000 burst)"

# Step 6: Configure method-level throttling for /login
echo ""
echo "[6] Configuring stricter throttling for /login..."
aws apigateway update-stage \
  --rest-api-id $API_ID \
  --stage-name prod \
  --patch-operations \
    "op=replace,path=/~1login/POST/throttling/rateLimit,value=10" \
    "op=replace,path=/~1login/POST/throttling/burstLimit,value=20" \
  2>/dev/null && \
  echo "✅ /login throttling: 10 RPS, burst 20"

# Step 7: Apply resource policy
echo ""
echo "[7] Applying resource policy to API..."
cat > /tmp/api-resource-policy.json << EOF
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Sid": "AllowAllTraffic",
      "Effect": "Allow",
      "Principal": "*",
      "Action": "execute-api:Invoke",
      "Resource": "arn:aws:execute-api:${REGION}:${ACCOUNT_ID}:${API_ID}/*"
    },
    {
      "Sid": "DenyBlockedIPs",
      "Effect": "Deny",
      "Principal": "*",
      "Action": "execute-api:Invoke",
      "Resource": "arn:aws:execute-api:${REGION}:${ACCOUNT_ID}:${API_ID}/*",
      "Condition": {
        "IpAddress": {
          "aws:SourceIp": [
            "192.0.2.0/24",
            "198.51.100.0/24"
          ]
        }
      }
    }
  ]
}
EOF

aws apigateway update-rest-api \
  --rest-api-id $API_ID \
  --patch-operations \
    op=replace,path=/policy,\
value=$(python3 -c \
  "import json,urllib.parse; \
  print(urllib.parse.quote(open('/tmp/api-resource-policy.json').read()))" \
  2>/dev/null) \
  2>/dev/null && \
  echo "✅ Resource policy applied" || \
  echo "ℹ️  Resource policy reference created"
echo "   Blocks: 192.0.2.0/24, 198.51.100.0/24"

# Step 8: Create API Key and Usage Plan
echo ""
echo "[8] Creating API Key and Usage Plan..."

# Create API Key
API_KEY_ID=$(aws apigateway create-api-key \
  --name "Lab104-PremiumKey" \
  --description "Lab104 premium customer API key" \
  --enabled \
  --tags Lab=Lab104 \
  --query 'id' \
  --output text 2>/dev/null)

API_KEY_VALUE=$(aws apigateway get-api-key \
  --api-key $API_KEY_ID \
```
