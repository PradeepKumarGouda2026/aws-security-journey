## Part 2 — Explore Identity Store (Users & Groups)

```bash
#!/bin/bash
# Lab 97 - Part 2: Identity Store Exploration
# Day 32 - AWS Security Specialist Study Plan

source /tmp/lab97-env.sh

echo "=============================================="
echo "  Part 2: Identity Store — Users & Groups"
echo "=============================================="

# Step 1: List all users
echo ""
echo "[1] Listing all users in Identity Store..."
aws identitystore list-users \
  --identity-store-id $IDENTITY_STORE_ID \
  --query 'Users[*].{
    UserId:UserId,
    UserName:UserName,
    DisplayName:DisplayName,
    Email:Emails[0].Value
  }' \
  --output table 2>/dev/null || echo "No users found or insufficient permissions"

# Step 2: List all groups
echo ""
echo "[2] Listing all groups in Identity Store..."
aws identitystore list-groups \
  --identity-store-id $IDENTITY_STORE_ID \
  --query 'Groups[*].{
    GroupId:GroupId,
    DisplayName:DisplayName,
    Description:Description
  }' \
  --output table 2>/dev/null || echo "No groups found or insufficient permissions"

# Step 3: Get a specific user's details (if users exist)
echo ""
echo "[3] Getting first user's full details..."
FIRST_USER_ID=$(aws identitystore list-users \
  --identity-store-id $IDENTITY_STORE_ID \
  --query 'Users[0].UserId' \
  --output text 2>/dev/null)

if [ "$FIRST_USER_ID" != "None" ] && [ ! -z "$FIRST_USER_ID" ]; then
  aws identitystore describe-user \
    --identity-store-id $IDENTITY_STORE_ID \
    --user-id $FIRST_USER_ID \
    --output json | python3 -m json.tool
else
  echo "No users found in identity store"
fi

# Step 4: Check group memberships
echo ""
echo "[4] Checking group memberships..."
FIRST_GROUP_ID=$(aws identitystore list-groups \
  --identity-store-id $IDENTITY_STORE_ID \
  --query 'Groups[0].GroupId' \
  --output text 2>/dev/null)

if [ "$FIRST_GROUP_ID" != "None" ] && [ ! -z "$FIRST_GROUP_ID" ]; then
  echo "Members of first group ($FIRST_GROUP_ID):"
  aws identitystore list-group-memberships \
    --identity-store-id $IDENTITY_STORE_ID \
    --group-id $FIRST_GROUP_ID \
    --query 'GroupMemberships[*].{MembershipId:MembershipId, MemberId:MemberId.UserId}' \
    --output table
else
  echo "No groups found in identity store"
fi

# Step 5: Create a sample user (for exploration)
echo ""
echo "[5] Creating sample lab user..."
aws identitystore create-user \
  --identity-store-id $IDENTITY_STORE_ID \
  --user-name "lab97.testuser" \
  --display-name "Lab97 Test User" \
  --name '{"GivenName":"Lab97","FamilyName":"TestUser"}' \
  --emails '[{"Value":"lab97testuser@example.com","Type":"work","Primary":true}]' \
  --query 'UserId' \
  --output text 2>/dev/null && echo "✅ Test user created" || \
  echo "⚠️  Could not create user (may already exist or insufficient permissions)"

# Step 6: Create a sample group
echo ""
echo "[6] Creating sample lab group..."
aws identitystore create-group \
  --identity-store-id $IDENTITY_STORE_ID \
  --display-name "Lab97-SecurityTeam" \
  --description "Lab 97 test group for security team" \
  --query 'GroupId' \
  --output text 2>/dev/null && echo "✅ Test group created" || \
  echo "⚠️  Could not create group (may already exist or insufficient permissions)"

echo ""
echo "=============================================="
echo "Part 2 Complete!"
echo "Key takeaway: Identity Store holds all users"
echo "and groups regardless of identity source."
echo "=============================================="
```

## Part 3 — Create & Inspect Permission Sets

```bash
#!/bin/bash
# Lab 97 - Part 3: Permission Sets
# Day 32 - AWS Security Specialist Study Plan

source /tmp/lab97-env.sh

echo "=============================================="
echo "  Part 3: Permission Sets"
echo "=============================================="

# Step 1: List existing permission sets
echo ""
echo "[1] Listing all existing Permission Sets..."
PERMISSION_SET_ARNS=$(aws sso-admin list-permission-sets \
  --instance-arn $INSTANCE_ARN \
  --query 'PermissionSets[]' \
  --output text 2>/dev/null)

if [ ! -z "$PERMISSION_SET_ARNS" ]; then
  for PS_ARN in $PERMISSION_SET_ARNS; do
    aws sso-admin describe-permission-set \
      --instance-arn $INSTANCE_ARN \
      --permission-set-arn $PS_ARN \
      --query 'PermissionSet.{Name:Name, SessionDuration:SessionDuration, Description:Description}' \
      --output table
  done
else
  echo "No existing permission sets found"
fi

# Step 2: Create SecurityAudit Permission Set
echo ""
echo "[2] Creating SecurityAudit Permission Set..."
SECURITY_AUDIT_PS_ARN=$(aws sso-admin create-permission-set \
  --instance-arn $INSTANCE_ARN \
  --name "Lab97-SecurityAudit" \
  --description "Read-only access to security services for audit team" \
  --session-duration "PT4H" \
  --relay-state "https://console.aws.amazon.com/securityhub/home" \
  --query 'PermissionSet.PermissionSetArn' \
  --output text 2>/dev/null)

if [ ! -z "$SECURITY_AUDIT_PS_ARN" ]; then
  echo "✅ SecurityAudit Permission Set created"
  echo "   ARN: $SECURITY_AUDIT_PS_ARN"
  echo "export SECURITY_AUDIT_PS_ARN=\"$SECURITY_AUDIT_PS_ARN\"" >> /tmp/lab97-env.sh
else
  echo "⚠️  Could not create permission set"
fi

# Step 3: Attach AWS managed policy to permission set
echo ""
echo "[3] Attaching SecurityAudit managed policy..."
aws sso-admin attach-managed-policy-to-permission-set \
  --instance-arn $INSTANCE_ARN \
  --permission-set-arn $SECURITY_AUDIT_PS_ARN \
  --managed-policy-arn "arn:aws:iam::aws:policy/SecurityAudit" 2>/dev/null && \
  echo "✅ SecurityAudit policy attached" || \
  echo "⚠️  Could not attach policy"

# Step 4: Add inline policy to permission set
echo ""
echo "[4] Adding inline policy (deny destructive actions)..."
cat > /tmp/ps-inline-policy.json << 'EOF'
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Sid": "DenyDestructiveActions",
      "Effect": "Deny",
      "Action": [
        "cloudtrail:DeleteTrail",
        "cloudtrail:StopLogging",
        "config:DeleteConfigRule",
        "guardduty:DeleteDetector",
        "securityhub:DisableSecurityHub"
      ],
      "Resource": "*"
    }
  ]
}
EOF

aws sso-admin put-inline-policy-to-permission-set \
  --instance-arn $INSTANCE_ARN \
  --permission-set-arn $SECURITY_AUDIT_PS_ARN \
  --inline-policy file:///tmp/ps-inline-policy.json 2>/dev/null && \
  echo "✅ Inline policy attached" || \
  echo "⚠️  Could not attach inline policy"

# Step 5: Create PowerUser Permission Set
echo ""
echo "[5] Creating PowerUser Permission Set..."
POWER_USER_PS_ARN=$(aws sso-admin create-permission-set \
  --instance-arn $INSTANCE_ARN \
  --name "Lab97-PowerUser" \
  --description "Power user access for developers" \
  --session-duration "PT8H" \
  --query 'PermissionSet.PermissionSetArn' \
  --output text 2>/dev/null)

if [ ! -z "$POWER_USER_PS_ARN" ]; then
  echo "✅ PowerUser Permission Set created"
  echo "   ARN: $POWER_USER_PS_ARN"
  echo "export POWER_USER_PS_ARN=\"$POWER_USER_PS_ARN\"" >> /tmp/lab97-env.sh

  # Attach PowerUserAccess managed policy
  aws sso-admin attach-managed-policy-to-permission-set \
    --instance-arn $INSTANCE_ARN \
    --permission-set-arn $POWER_USER_PS_ARN \
    --managed-policy-arn "arn:aws:iam::aws:policy/PowerUserAccess" 2>/dev/null && \
    echo "✅ PowerUserAccess policy attached"
fi

# Step 6: Create BreakGlass Permission Set
echo ""
echo "[6] Creating BreakGlass Permission Set (short session)..."
BREAK_GLASS_PS_ARN=$(aws sso-admin create-permission-set \
  --instance-arn $INSTANCE_ARN \
  --name "Lab97-BreakGlass" \
  --description "Emergency admin access - 1 hour session only" \
  --session-duration "PT1H" \
  --query 'PermissionSet.PermissionSetArn' \
  --output text 2>/dev/null)

if [ ! -z "$BREAK_GLASS_PS_ARN" ]; then
  echo "✅ BreakGlass Permission Set created (1hr session)"
  echo "   ARN: $BREAK_GLASS_PS_ARN"
  echo "export BREAK_GLASS_PS_ARN=\"$BREAK_GLASS_PS_ARN\"" >> /tmp/lab97-env.sh

  aws sso-admin attach-managed-policy-to-permission-set \
    --instance-arn $INSTANCE_ARN \
    --permission-set-arn $BREAK_GLASS_PS_ARN \
    --managed-policy-arn "arn:aws:iam::aws:policy/AdministratorAccess" 2>/dev/null && \
    echo "✅ AdministratorAccess policy attached to BreakGlass"
fi

# Step 7: Verify permission set details
echo ""
echo "[7] Verifying SecurityAudit Permission Set details..."
aws sso-admin describe-permission-set \
  --instance-arn $INSTANCE_ARN \
  --permission-set-arn $SECURITY_AUDIT_PS_ARN \
  --output json | python3 -m json.tool 2>/dev/null

echo ""
echo "[8] Listing policies attached to SecurityAudit..."
aws sso-admin list-managed-policies-in-permission-set \
  --instance-arn $INSTANCE_ARN \
  --permission-set-arn $SECURITY_AUDIT_PS_ARN \
  --query 'AttachedManagedPolicies[*].{Name:Name, Arn:Arn}' \
  --output table 2>/dev/null

echo ""
echo "[9] Getting inline policy of SecurityAudit..."
aws sso-admin get-inline-policy-for-permission-set \
  --instance-arn $INSTANCE_ARN \
  --permission-set-arn $SECURITY_AUDIT_PS_ARN \
  --query 'InlinePolicy' \
  --output text 2>/dev/null | python3 -m json.tool

echo ""
echo "=============================================="
echo "Part 3 Complete!"
echo "Three permission sets created:"
echo "  1. Lab97-SecurityAudit  (4hr session)"
echo "  2. Lab97-PowerUser      (8hr session)"
echo "  3. Lab97-BreakGlass     (1hr session)"
echo "=============================================="
```

## Part 4 — Account Assignments & Provisioning

```bash
#!/bin/bash
# Lab 97 - Part 4: Account Assignments
# Day 32 - AWS Security Specialist Study Plan

source /tmp/lab97-env.sh

echo "=============================================="
echo "  Part 4: Account Assignments & Provisioning"
echo "=============================================="

# Get current account ID
ACCOUNT_ID=$(aws sts get-caller-identity --query Account --output text)
echo "Working with Account ID: $ACCOUNT_ID"

# Step 1: Check existing account assignments
echo ""
echo "[1] Checking existing account assignments..."
aws sso-admin list-permission-sets-provisioned-to-account \
  --instance-arn $INSTANCE_ARN \
  --account-id $ACCOUNT_ID \
  --query 'PermissionSets[]' \
  --output table 2>/dev/null || echo "No assignments found"

# Step 2: Get group ID for assignment
echo ""
echo "[2] Getting Lab97-SecurityTeam group ID..."
SECURITY_GROUP_ID=$(aws identitystore list-groups \
  --identity-store-id $IDENTITY_STORE_ID \
  --filters '[{"AttributePath":"DisplayName","AttributeValue":"Lab97-SecurityTeam"}]' \
  --query 'Groups[0].GroupId' \
  --output text 2>/dev/null)

echo "Security Team Group ID: $SECURITY_GROUP_ID"

# Step 3: Create account assignment (Group → Permission Set → Account)
echo ""
echo "[3] Assigning SecurityAudit permission set to SecurityTeam group..."
if [ ! -z "$SECURITY_GROUP_ID" ] && [ "$SECURITY_GROUP_ID" != "None" ]; then
  aws sso-admin create-account-assignment \
    --instance-arn $INSTANCE_ARN \
    --target-id $ACCOUNT_ID \
    --target-type AWS_ACCOUNT \
    --permission-set-arn $SECURITY_AUDIT_PS_ARN \
    --principal-type GROUP \
    --principal-id $SECURITY_GROUP_ID \
    --output json | python3 -m json.tool 2>/dev/null && \
    echo "✅ Account assignment created" || \
    echo "⚠️  Could not create assignment"
else
  echo "⚠️  Group not found — skipping assignment"
fi

# Step 4: Provision permission set to account
echo ""
echo "[4] Provisioning SecurityAudit Permission Set to account..."
aws sso-admin provision-permission-set \
  --instance-arn $INSTANCE_ARN \
  --permission-set-arn $SECURITY_AUDIT_PS_ARN \
  --target-type AWS_ACCOUNT \
  --target-id $ACCOUNT_ID \
  --output json 2>/dev/null && \
  echo "✅ Permission set provisioned" || \
  echo "⚠️  Could not provision (may need management account)"

# Step 5: List assignments for the permission set
echo ""
echo "[5] Listing all assignments for SecurityAudit Permission Set..."
aws sso-admin list-account-assignments \
  --instance-arn $INSTANCE_ARN \
  --account-id $ACCOUNT_ID \
  --permission-set-arn $SECURITY_AUDIT_PS_ARN \
  --query 'AccountAssignments[*].{
    AccountId:AccountId,
    PrincipalType:PrincipalType,
    PrincipalId:PrincipalId
  }' \
  --output table 2>/dev/null

# Step 6: Verify IAM role was created in account
echo ""
echo "[6] Verifying AWSReservedSSO role was created..."
aws iam list-roles \
  --query 'Roles[?starts_with(RoleName, `AWSReservedSSO`)].{
    RoleName:RoleName,
    CreateDate:CreateDate,
    RoleId:RoleId
  }' \
  --output table 2>/dev/null

# Step 7: Inspect the auto-created role
echo ""
echo "[7] Inspecting auto-created SSO role trust policy..."
SSO_ROLE_NAME=$(aws iam list-roles \
  --query 'Roles[?starts_with(RoleName, `AWSReservedSSO_Lab97-SecurityAudit`)].RoleName' \
  --output text 2>/dev/null)

if [ ! -z "$SSO_ROLE_NAME" ]; then
  echo "Role Name: $SSO_ROLE_NAME"
  aws iam get-role \
    --role-name $SSO_ROLE_NAME \
    --query 'Role.AssumeRolePolicyDocument' \
    --output json | python3 -m json.tool
else
  echo "⚠️  SSO role not yet created or insufficient permissions"
fi

echo ""
echo "=============================================="
echo "Part 4 Complete!"
echo "Key takeaway: IAM Identity Center automatically"
echo "creates and manages IAM roles in target accounts."
echo "Never modify AWSReservedSSO_* roles manually!"
echo "=============================================="
```

## Part 6 — CloudTrail Audit for IAM Identity Center

```bash
#!/bin/bash
# Lab 97 - Part 6: CloudTrail Audit
# Day 32 - AWS Security Specialist Study Plan

source /tmp/lab97-env.sh

echo "=============================================="
echo "  Part 6: CloudTrail Audit — Identity Center"
echo "=============================================="

# IAM Identity Center CloudTrail events are in us-east-1
AUDIT_REGION="us-east-1"

# Step 1: Search for Permission Set events
echo ""
echo "[1] Searching CloudTrail for Permission Set creation events..."
aws cloudtrail lookup-events \
  --region $AUDIT_REGION \
  --lookup-attributes AttributeKey=EventName,AttributeValue=CreatePermissionSet \
  --start-time $(date -u -d '1 hour ago' '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null || \
                 date -u -v-1H '+%Y-%m-%dT%H:%M:%SZ') \
  --query 'Events[*].{
    EventTime:EventTime,
    EventName:EventName,
    Username:Username,
    EventId:EventId
  }' \
  --output table 2>/dev/null || echo "No recent CreatePermissionSet events found"

# Step 2: Search for account assignment events
echo ""
echo "[2] Searching CloudTrail for Account Assignment events..."
aws cloudtrail lookup-events \
  --region $AUDIT_REGION \
  --lookup-attributes AttributeKey=EventName,AttributeValue=CreateAccountAssignment \
  --start-time $(date -u -d '1 hour ago' '+%Y-%m-%dT%H:%M:%SZ' 2>/dev/null || \
                 date -u -v-1H '+%Y-%m-%dT%H:%M:%SZ') \
  --query 'Events[*].{
    EventTime:EventTime,
    EventName:EventName,
    Username:Username
  }' \
  --output table 2>/dev/null || echo "No recent CreateAccountAssignment events found"

# Step 3: Search for SSO authentication events
echo ""
echo "[3] Searching CloudTrail for SSO authentication events..."
aws cloudtrail lookup-events \
  --region $AUDIT_REGION \
  --lookup-attributes AttributeKey=EventSource,AttributeValue=sso.amazonaws.com \
  --max-results 10 \
  --query 'Events[*].{
    EventTime:EventTime,
    EventName:EventName,
    Username:Username,
    SourceIP:CloudTrailEvent
  }' \
  --output table 2>/dev/null || echo "No recent SSO events found"

# Step 4: Create EventBridge rule for Identity Center monitoring
echo ""
echo "[4] Creating EventBridge rule for Permission Set changes..."
aws events put-rule \
  --name "Lab97-IdentityCenter-PermissionSetChanges" \
  --event-pattern '{
    "source": ["aws.sso-admin"],
    "detail-type": ["AWS API Call via CloudTrail"],
    "detail": {
      "eventSource": ["sso-admin.amazonaws.com"],
      "eventName": [
        "CreatePermissionSet",
        "DeletePermissionSet",
        "CreateAccountAssignment",
        "DeleteAccountAssignment",
        "ProvisionPermissionSet"
      ]
    }
  }' \
  --state ENABLED \
  --description "Alert on IAM Identity Center permission set and assignment changes" \
  --region $AUDIT_REGION \
  --output table 2>/dev/null && \
  echo "✅ EventBridge rule created for Identity Center monitoring" || \
  echo "⚠️  Could not create EventBridge rule"

# Step 5: Show full audit summary
echo ""
echo "[5] Full Identity Center audit summary..."
echo ""
echo "Events to monitor in CloudTrail (source: sso-admin.amazonaws.com):"
echo "┌──────────────────────────────────────┬──────────────────────────────────┐"
echo "│ Event Name                           │ Security Significance            │"
echo "├──────────────────────────────────────┼──────────────────────────────────┤"
echo "│ CreateAccountAssignment              │ New access granted               │"
```

```bash
#!/bin/bash
# Lab 97 - Part 7: Cleanup

    --instance-arn $INSTANCE_ARN \

    --permission-set-arn $SECURITY_AUDIT_PS_ARN \

    echo "✅ Account assignment removed" || echo "⚠️  Assignment already removed"

  if [ ! -z "$PS_ARN" ] && [ "$PS_ARN" != "None" ]; then
    aws sso-admin delete-permission-set \
      --instance-arn $INSTANCE_ARN \
      --permission-set-arn $PS_ARN 2>/dev/null && \

      echo "⚠️  Could not delete: $PS_ARN"

    --identity-store-id $IDENTITY_STORE_ID \

echo "All lab resources removed."
```
