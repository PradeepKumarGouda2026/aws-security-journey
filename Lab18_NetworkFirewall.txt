### **Step 1: Create VPC Infrastructure (15 minutes)**

```bash
#!/bin/bash
# CloudShell - Network Firewall Lab Setup

echo "╔═══════════════════════════════════════════════════════════╗"
echo "║   AWS NETWORK FIREWALL - COMPLETE LAB SETUP              ║"
echo "╚═══════════════════════════════════════════════════════════╝"
echo ""

# Variables for consistent naming
TIMESTAMP=$(date +%s)
REGION="us-east-1"

echo "Step 1: Creating VPC Infrastructure"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"

# Create VPC
echo "  → Creating VPC (10.3.0.0/16)..."
NFW_VPC=$(aws ec2 create-vpc \
  --cidr-block 10.3.0.0/16 \
  --tag-specifications 'ResourceType=vpc,Tags=[{Key=Name,Value=NetworkFirewall-VPC}]' \
  --query 'Vpc.VpcId' \
  --output text)

echo "     ✅ VPC created: $NFW_VPC"

# Enable DNS
aws ec2 modify-vpc-attribute --vpc-id $NFW_VPC --enable-dns-hostnames
aws ec2 modify-vpc-attribute --vpc-id $NFW_VPC --enable-dns-support
echo "     ✅ DNS hostname and support enabled"

# Create Firewall Subnet (where firewall endpoint lives)
echo ""
echo "  → Creating Firewall Subnet (10.3.1.0/24) in us-east-1a..."
NFW_FIREWALL_SUBNET=$(aws ec2 create-subnet \
  --vpc-id $NFW_VPC \
  --cidr-block 10.3.1.0/24 \
  --availability-zone us-east-1a \
  --tag-specifications 'ResourceType=subnet,Tags=[{Key=Name,Value=Firewall-Subnet}]' \
  --query 'Subnet.SubnetId' \
  --output text)

echo "     ✅ Firewall subnet created: $NFW_FIREWALL_SUBNET"
echo "     ℹ️  This subnet hosts the firewall endpoint only"

# Create Protected Subnet (your workloads)
echo ""
echo "  → Creating Protected Subnet (10.3.2.0/24) in us-east-1a..."
NFW_PROTECTED_SUBNET=$(aws ec2 create-subnet \
  --vpc-id $NFW_VPC \
  --cidr-block 10.3.2.0/24 \
  --availability-zone us-east-1a \
  --tag-specifications 'ResourceType=subnet,Tags=[{Key=Name,Value=Protected-Subnet}]' \
  --query 'Subnet.SubnetId' \
  --output text)

# Enable auto-assign public IP
aws ec2 modify-subnet-attribute \
  --subnet-id $NFW_PROTECTED_SUBNET \
  --map-public-ip-on-launch

echo "     ✅ Protected subnet created: $NFW_PROTECTED_SUBNET"
echo "     ✅ Auto-assign public IP enabled"
echo "     ℹ️  Your EC2 instances will go here"

# Create Internet Gateway
echo ""
echo "  → Creating Internet Gateway..."
NFW_IGW=$(aws ec2 create-internet-gateway \
  --tag-specifications 'ResourceType=internet-gateway,Tags=[{Key=Name,Value=NFW-IGW}]' \
  --query 'InternetGateway.InternetGatewayId' \
  --output text)

# Attach IGW to VPC
aws ec2 attach-internet-gateway \
  --internet-gateway-id $NFW_IGW \
  --vpc-id $NFW_VPC

echo "     ✅ Internet Gateway created and attached: $NFW_IGW"

echo ""
echo "✅ VPC Infrastructure Complete!"
echo ""
echo "Summary:"
echo "  VPC:              $NFW_VPC (10.3.0.0/16)"
echo "  Firewall Subnet:  $NFW_FIREWALL_SUBNET (10.3.1.0/24)"
echo "  Protected Subnet: $NFW_PROTECTED_SUBNET (10.3.2.0/24)"
echo "  Internet Gateway: $NFW_IGW"
echo ""
```

### **Step 2: Create Stateless Rule Group (10 minutes)**

```bash
#!/bin/bash
# CloudShell - Stateless Rule Group Creation

echo "Step 2: Creating Stateless Rule Group"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo ""
echo "Stateless rules are evaluated FIRST (before stateful)"
echo "They're fast packet filters with priority-based evaluation"
echo ""

# Create stateless rules JSON
cat > /tmp/stateless-rules.json << 'EOF'
{
  "RulesSource": {
    "StatelessRulesAndCustomActions": {
      "StatelessRules": [
        {
          "RuleDefinition": {
            "MatchAttributes": {
              "Sources": [{"AddressDefinition": "198.51.100.0/24"}],
              "Destinations": [{"AddressDefinition": "0.0.0.0/0"}]
            },
            "Actions": ["aws:drop"]
          },
          "Priority": 1
        },
        {
          "RuleDefinition": {
            "MatchAttributes": {
              "Sources": [{"AddressDefinition": "203.0.113.0/24"}],
              "Destinations": [{"AddressDefinition": "0.0.0.0/0"}]
            },
            "Actions": ["aws:drop"]
          },
          "Priority": 2
        },
        {
          "RuleDefinition": {
            "MatchAttributes": {
              "Sources": [{"AddressDefinition": "0.0.0.0/0"}],
              "Destinations": [{"AddressDefinition": "0.0.0.0/0"}],
              "DestinationPorts": [{"FromPort": 80, "ToPort": 80}],
              "Protocols": [6]
            },
            "Actions": ["aws:forward_to_sfe"]
          },
          "Priority": 10
        },
        {
          "RuleDefinition": {
            "MatchAttributes": {
              "Sources": [{"AddressDefinition": "0.0.0.0/0"}],
              "Destinations": [{"AddressDefinition": "0.0.0.0/0"}],
              "DestinationPorts": [{"FromPort": 443, "ToPort": 443}],
              "Protocols": [6]
            },
            "Actions": ["aws:forward_to_sfe"]
          },
          "Priority": 11
        },
        {
          "RuleDefinition": {
            "MatchAttributes": {
              "Sources": [{"AddressDefinition": "0.0.0.0/0"}],
              "Destinations": [{"AddressDefinition": "0.0.0.0/0"}]
            },
            "Actions": ["aws:forward_to_sfe"]
          },
          "Priority": 100
        }
      ]
    }
  }
}
EOF

echo "  → Creating stateless rule group..."
echo ""
echo "     Rules being created:"
echo "       Priority 1:  DROP 198.51.100.0/24 (malicious IP range)"
echo "       Priority 2:  DROP 203.0.113.0/24 (another bad range)"
echo "       Priority 10: FORWARD port 80 (HTTP) to stateful inspection"
echo "       Priority 11: FORWARD port 443 (HTTPS) to stateful inspection"
echo "       Priority 100: FORWARD all other traffic to stateful inspection"
echo ""

NFW_STATELESS_RG=$(aws network-firewall create-rule-group \
  --rule-group-name StatelessFilteringRules \
  --type STATELESS \
  --capacity 100 \
  --rule-group file:///tmp/stateless-rules.json \
  --description "Fast packet filtering - blocks known bad IPs, forwards rest to stateful" \
  --query 'RuleGroupResponse.RuleGroupArn' \
  --output text)

echo "     ✅ Stateless rule group created!"
echo "     ARN: $NFW_STATELESS_RG"
echo "     Capacity: 100 units allocated"
echo ""
echo "     How it works:"
echo "       1. Packet arrives"
echo "       2. Check priority 1: Is it from 198.51.100.0/24? → DROP"
echo "       3. Check priority 2: Is it from 203.0.113.0/24? → DROP"
echo "       4. Check priority 10: Is it HTTP (port 80)? → FORWARD to stateful"
echo "       5. Check priority 11: Is it HTTPS (port 443)? → FORWARD to stateful"
echo "       6. Priority 100: Everything else → FORWARD to stateful"
echo ""
```

cat > /tmp/suricata-rules.json << 'EOF'
{
  "RulesSource": {

### **Step 7: Deploy Network Firewall (15 minutes)**

```bash
#!/bin/bash
# CloudShell - Deploy Network Firewall

echo "Step 7: Deploying Network Firewall"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo ""
echo "This will:"
echo "  • Create firewall endpoint in firewall subnet"
echo "  • Apply comprehensive security policy"
echo "  • Enable logging"
echo "  • Takes 3-5 minutes to provision"
echo ""

echo "  → Creating Network Firewall..."
echo ""

NFW_FIREWALL_ARN=$(aws network-firewall create-firewall \
  --firewall-name ProductionFirewall \
  --firewall-policy-arn $NFW_POLICY_ARN \
  --vpc-id $NFW_VPC \
  --subnet-mappings SubnetId=$NFW_FIREWALL_SUBNET \
  --description "Production Network Firewall with comprehensive security rules" \
  --tags Key=Name,Value=Production-Firewall Key=Environment,Value=Lab \
  --query 'Firewall.FirewallArn' \
  --output text)

# Extract firewall ID
NFW_FIREWALL_ID=$(echo $NFW_FIREWALL_ARN | awk -F'/' '{print $NF}')

echo "     ✅ Firewall creation initiated!"
echo "     ARN: $NFW_FIREWALL_ARN"
echo "     ID: $NFW_FIREWALL_ID"
echo ""
echo "  → Waiting for firewall to become ready..."
echo "     This takes 3-5 minutes. Please wait..."
echo ""

# Function to check firewall status
check_firewall_status() {
    STATUS=$(aws network-firewall describe-firewall \
        --firewall-arn $NFW_FIREWALL_ARN \
        --query 'FirewallStatus.Status' \
        --output text 2>/dev/null)
    echo $STATUS
}

# Wait loop with progress indicator
COUNTER=0

        aws ec2 detach-internet-gateway --internet-gateway-id $NFW_IGW --vpc-id $NFW_VPC 2>/dev/null
        aws ec2 delete-internet-gateway --internet-gateway-id $NFW_IGW 2>/dev/null
        echo "  ✅ Internet Gateway deleted"
    fi
    
    # Delete route tables
    for rt in $NFW_IGW_RT $NFW_PROTECTED_RT $NFW_FIREWALL_RT; do
        if [ ! -z "$rt" ]; then
            # Disassociate
            ASSOCS=$(aws ec2 describe-route-tables \
                --route-table-ids $rt \
                --query 'RouteTables[0].Associations[?SubnetId!=`null`].RouteTableAssociationId' \
                --output text 2>/dev/null)

                aws ec2 disassociate-route-table --association-id $assoc 2>/dev/null
            done
            
            aws ec2 delete-route-table --route-table-id $rt 2>/dev/null
        fi
    done
    echo "  ✅ Route tables deleted"
    
    # Delete subnets
    for subnet in $NFW_FIREWALL_SUBNET $NFW_PROTECTED_SUBNET; do
        if [ ! -z "$subnet" ]; then
            aws ec2 delete-subnet --subnet-id $subnet 2>/dev/null
        fi
    done
    echo "  ✅ Subnets deleted"

    aws ec2 delete-vpc --vpc-id $NFW_VPC 2>/dev/null

## Components Deep Dive

### 1. Stateless Rules
- **Evaluated FIRST** (before stateful)
- **No connection tracking**
- **Priority-based** (1-65535, lower = higher priority)
- **Fast packet filtering**

**Actions:**
- `aws:pass` - Allow immediately, skip stateful
- `aws:drop` - Drop packet immediately
- `aws:forward_to_sfe` - Send to stateful engine

**Use Cases:**
- Quick DROP of known malicious IPs
- PASS trusted internal traffic (performance optimization)
- Rate limiting with custom actions

**Example Rules:**
```
Priority 1: DROP 198.51.100.0/24 (malicious IPs)
Priority 10: FORWARD port 80,443 to stateful
Priority 100: FORWARD all other to stateful
```

---

### 2. Stateful Rules (Three Types)

#### A) Standard 5-Tuple Rules

**Format:** Protocol, Source IP/Port, Dest IP/Port, Direction

**Actions:**
- `PASS` - Allow traffic
- `DROP` - Block silently
- `REJECT` - Block with TCP reset/ICMP unreachable
- `ALERT` - Log only (IDS mode)

**Example:**
```json
{
  "Action": "DROP",
  "Header": {
    "Protocol": "TCP",
    "Source": "ANY",
    "SourcePort": "ANY",
    "Direction": "FORWARD",
    "Destination": "ANY",
    "DestinationPort": "22"
  },
  "RuleOptions": [{"Keyword": "sid:1000001"}]
}
```

**Use:** Block specific ports (SSH, RDP, Telnet)

---

#### B) Domain Filtering

**Purpose:** Filter by domain name (not just IP!)

**How It Works:**
- Inspects HTTP `Host:` header
- Inspects TLS SNI (Server Name Indication)
- Supports wildcards (`.facebook.com` = all subdomains)

**Rule Types:**
- `ALLOWLIST` - Only listed domains allowed, block all else
- `DENYLIST` - Block listed domains, allow all else

**Example:**
```json
{
  "RulesSourceList": {
    "TargetTypes": ["HTTP_HOST", "TLS_SNI"],
    "Targets": [
      ".facebook.com",
      ".twitter.com",
      ".instagram.com"
    ],
    "GeneratedRulesType": "DENYLIST"
  }
}
```

**Use Cases:**
- Corporate internet filtering (block social media)
- Compliance (block gambling, adult content)
- Malware prevention (block known C2 domains)

---

#### C) Suricata IDS/IPS Rules

**Suricata** = Open-source IDS/IPS engine with deep packet inspection

**Rule Syntax:**
```
action protocol src_ip src_port direction dst_ip dst_port (options)
```

**Example Rules Created:**

```
SID 1000010: SQL Injection - "UNION SELECT"
alert http any any -> any any (
  msg:"SQL Injection Detected";
  flow:to_server,established;
  content:"UNION SELECT";
  nocase;
  sid:1000010;
  rev:1;
)

SID 1000012: XSS - "<script>" tag
alert http any any -> any any (
  msg:"XSS Attack Detected";
  flow:to_server,established;
  content:"<script>";
  nocase;
  sid:1000012;
  rev:1;
)

SID 1000014: Path Traversal - "../"
alert http any any -> any any (
  msg:"Path Traversal Attack";
  flow:to_server,established;
  content:"..%2f";
  nocase;
  sid:1000014;
  rev:1;
)
```

**Rule Options:**
- `msg` - Alert message
- `flow` - Connection state
- `content` - Pattern to match
- `pcre` - Regex pattern
- `sid` - Signature ID (must be unique)
- `nocase` - Case-insensitive matching

**IDS vs IPS Mode:**
- `alert` - IDS mode (log only, don't block)
- `drop` - IPS mode (block traffic)

**Best Practice:**
1. Start with `alert` (IDS)
2. Monitor logs for false positives
3. Tune rules
4. Switch to `drop` (IPS) after validation

## Traffic Flow

**Inbound (Internet → EC2):**
```
1. Packet arrives at Internet Gateway
2. IGW routes to firewall endpoint (via route table)
3. Firewall evaluates:
   a. Stateless rules (quick filter)
   b. Stateful rules (deep inspection)
   c. Domain filtering
   d. Suricata IDS
4. Decision: ALLOW or DROP
5. If ALLOW: Forwarded to protected subnet
6. EC2 receives packet
```

**Outbound (EC2 → Internet):**
```
1. EC2 sends packet
2. Protected subnet RT routes to firewall endpoint
3. Firewall inspects (same as inbound)
4. If ALLOW: Forwarded to IGW
5. IGW sends to internet
```

**ALL traffic inspected in BOTH directions!** 🔍

## Exam Tips

**What to Know:**

1. **Stateless vs Stateful Difference:**
   - Stateless: No connection tracking, both directions need rules, priority-based
   - Stateful: Connection tracking, automatic return traffic, actions include PASS/DROP/REJECT/ALERT

2. **When to Use Network Firewall vs Others:**
   - Security Groups: Instance-level, stateful, simple allow rules
   - NACLs: Subnet-level, stateless, allow + deny rules
   - Network Firewall: VPC-level, IDS/IPS, domain filtering, deep inspection
   - WAF: Application-layer (HTTP/HTTPS), web attacks, rate limiting

3. **Domain Filtering Limitations:**
   - HTTP/HTTPS only
   - Requires Host header or SNI
   - Encrypted SNI (ESNI) not supported

4. **Suricata IDS vs IPS:**
   - IDS: `alert` action, logs only
   - IPS: `drop` action, blocks traffic
   - Start with IDS, switch to IPS after tuning

5. **Cost Components:**
   - Endpoint: $0.395/hour/AZ (fixed)
   - Data: $0.065/GB (variable)
   - Minimum 2 AZ for HA

6. **Capacity Units:**
   - Set at creation, CANNOT be changed
   - Range: 100 to 30,000 per rule group
   - Plan ahead!

### What is AWS WAF?

## WAF Components

### 1. Web ACL (Access Control List)

**Container for all rules**

**Properties:**
- Name: Unique identifier
- Scope: REGIONAL (ALB, API Gateway) or CLOUDFRONT (global)
- Default Action: ALLOW or BLOCK
- Rules: Priority-ordered list
- Capacity: Max 1500 WCU (Web ACL Capacity Units)

**Default Action Strategy:**

**Approach 1: Default ALLOW** (Most Common)
- Allow everything EXCEPT what's blocked
- Rules define blocklist
- Good for: Public websites
- Example: Block only known attacks

**Approach 2: Default BLOCK** (Strict)
- Block everything EXCEPT what's allowed
- Rules define allowlist
- Good for: High-security APIs
- Example: Allow only specific patterns

---

### 2. Rules

**Three Types:**

#### A) Regular Rules
- Custom conditions you define
- Match statements (what to look for)
- Action (what to do if matched)

#### B) Rate-Based Rules
- Count requests per 5-minute window
- Threshold: 100 to 2,000,000,000
- Scope: Per IP or custom keys
- Action: Block automatically when exceeded

#### C) Managed Rule Groups
- Pre-built by AWS
- Maintained and updated automatically
- Cover OWASP Top 10
- Most are FREE!

---

### 3. Rule Actions

| Action | Description | HTTP Code |
|--------|-------------|-----------|
| ALLOW | Let request through | 200 (pass-through) |
| BLOCK | Block request | 403 Forbidden |
| COUNT | Count only (monitor) | 200 (pass-through) |
| CAPTCHA | Challenge with CAPTCHA | CAPTCHA page |
| Challenge | JavaScript challenge | JS challenge |

**Best Practice Workflow:**
1. Deploy in COUNT mode
2. Monitor for false positives (1-2 weeks)
3. Tune rules
4. Switch to CHALLENGE mode
5. Monitor user complaints
6. If acceptable, switch to BLOCK

---

### 4. Match Statements

**Available Match Types:**

**Geographic Match:**
- Match by country code (ISO 3166-1 alpha-2)
- Example: Block CN, RU, NK
- Accuracy: ~95-99%
- Limitation: VPN/proxy bypass

**IP Set Match:**
- Match against IP lists
- Up to 10,000 IPs per set
- Dynamic updates via API
- Use: Blocklist/allowlist

**Regex Pattern Set:**
- Regular expression matching
- Up to 10 patterns per set
- Use: Complex pattern detection
- Example: Credit card numbers

**Size Constraint:**
- Check request component size
- Operators: EQ, NE, GT, GE, LT, LE
- Use: Limit request body size

**SQL Injection Match:**
- AWS-managed signatures
- Detects SQL injection patterns
- Example: `UNION SELECT`

**XSS Match:**
- AWS-managed signatures
- Detects cross-site scripting
- Example: `<script>` tags

**String Match:**
- Simple string matching
- Case-sensitive or insensitive
- Position: EXACTLY, STARTS_WITH, ENDS_WITH, CONTAINS
- Transformations: LOWERCASE, URL_DECODE, HTML_DECODE

## Rate-Based Rules Deep Dive

**How They Work:**
1. Track requests per IP (or custom scope)
2. Count over 5-minute rolling window
3. If count > threshold: Take action
4. Block duration: ~10 minutes (auto-expires)

**Configuration:**

**Threshold:**
- Minimum: 100 requests/5 min
- Maximum: 2,000,000,000 requests/5 min

## Exam Tips

**What to Know:**

1. **WAF vs Network Firewall:**
   - WAF: Layer 7, web attacks, rate limiting, $10-50/month
   - Network Firewall: Layer 3-7, IDS/IPS, domain filtering, $600+/month

2. **Shield Standard vs Advanced:**
   - Standard: FREE, Layer 3/4, automatic, 99.99% effective
   - Advanced: $3k/month, Layer 7, DRT, cost protection, EC2 EIP

3. **AWS Managed Rules:**
   - Core Rule Set: OWASP Top 10 (FREE)
   - Bot Control: Only PAID managed rule ($10/month)
   - All others: FREE

4. **Rate-Based Rules:**
   - Threshold: 100 to 2 billion req/5min
   - Scope: IP, Forwarded IP, Custom Keys
   - Block duration: ~10 minutes

5. **Rule Actions:**
   - ALLOW: Let through
   - BLOCK: 403 Forbidden
   - COUNT: Monitor only
   - CAPTCHA: Challenge user
   - Challenge: JavaScript check

✅ WAF with IP Set (allowlist) + default BLOCK

✅ AWS Managed Rules Core Rule Set (FREE)

  --name MaliciousIPs \
  --scope REGIONAL \
  --ip-address-version IPV4 \
  --addresses "198.51.100.0/24" "203.0.113.0/24"

**Create Web ACL:**
```bash
aws wafv2 create-web-acl \
  --name MyWebACL \
  --scope REGIONAL \
  --default-action Allow={} \
  --rules file://rules.json \
  --visibility-config SampledRequestsEnabled=true,CloudWatchMetricsEnabled=true,MetricName=MyWebACL
```

```bash
aws wafv2 associate-web-acl \
  --web-acl-arn arn:aws:wafv2:... \
  --resource-arn arn:aws:elasticloadbalancing:...
```

```bash
aws wafv2 get-sampled-requests \

  --time-window StartTime=...,EndTime=... \

```
