### **Step 1: Create VPC Infrastructure for WAF Lab (15 minutes)**

```bash
#!/bin/bash
# CloudShell - WAF Lab Infrastructure Setup

echo "╔═══════════════════════════════════════════════════════════╗"
echo "║          AWS WAF - COMPLETE LAB SETUP                    ║"
echo "╚═══════════════════════════════════════════════════════════╝"
echo ""

echo "Step 1: Creating VPC Infrastructure for WAF Testing"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo ""

# Create VPC
echo "  → Creating VPC (10.4.0.0/16)..."
WAF_VPC=$(aws ec2 create-vpc \
  --cidr-block 10.4.0.0/16 \
  --tag-specifications 'ResourceType=vpc,Tags=[{Key=Name,Value=WAF-VPC}]' \
  --query 'Vpc.VpcId' \
  --output text)

echo "     ✅ VPC created: $WAF_VPC"

# Enable DNS
aws ec2 modify-vpc-attribute --vpc-id $WAF_VPC --enable-dns-hostnames
aws ec2 modify-vpc-attribute --vpc-id $WAF_VPC --enable-dns-support
echo "     ✅ DNS enabled"
echo ""

# Create subnets in TWO different AZs (ALB requires 2+ AZs)
echo "  → Creating subnets in multiple AZs..."
echo "     (ALB requires subnets in at least 2 Availability Zones)"
echo ""

WAF_SUBNET_1=$(aws ec2 create-subnet \
  --vpc-id $WAF_VPC \
  --cidr-block 10.4.1.0/24 \
  --availability-zone us-east-1a \
  --tag-specifications 'ResourceType=subnet,Tags=[{Key=Name,Value=WAF-Subnet-1}]' \
  --query 'Subnet.SubnetId' \
  --output text)

echo "     ✅ Subnet 1 created: $WAF_SUBNET_1 (us-east-1a)"

WAF_SUBNET_2=$(aws ec2 create-subnet \
  --vpc-id $WAF_VPC \
  --cidr-block 10.4.2.0/24 \
  --availability-zone us-east-1b \
  --tag-specifications 'ResourceType=subnet,Tags=[{Key=Name,Value=WAF-Subnet-2}]' \
  --query 'Subnet.SubnetId' \
  --output text)

echo "     ✅ Subnet 2 created: $WAF_SUBNET_2 (us-east-1b)"
echo ""

# Enable auto-assign public IP
aws ec2 modify-subnet-attribute --subnet-id $WAF_SUBNET_1 --map-public-ip-on-launch
aws ec2 modify-subnet-attribute --subnet-id $WAF_SUBNET_2 --map-public-ip-on-launch
echo "     ✅ Auto-assign public IP enabled on both subnets"
echo ""

# Create Internet Gateway
echo "  → Creating Internet Gateway..."
WAF_IGW=$(aws ec2 create-internet-gateway \
  --tag-specifications 'ResourceType=internet-gateway,Tags=[{Key=Name,Value=WAF-IGW}]' \
  --query 'InternetGateway.InternetGatewayId' \
  --output text)

# Attach to VPC
aws ec2 attach-internet-gateway --internet-gateway-id $WAF_IGW --vpc-id $WAF_VPC

echo "     ✅ Internet Gateway attached: $WAF_IGW"
echo ""

# Create route table
echo "  → Creating route table..."
WAF_RT=$(aws ec2 create-route-table \
  --vpc-id $WAF_VPC \
  --tag-specifications 'ResourceType=route-table,Tags=[{Key=Name,Value=WAF-RT}]' \
  --query 'RouteTable.RouteTableId' \
  --output text)

# Add route to internet
aws ec2 create-route --route-table-id $WAF_RT --destination-cidr-block 0.0.0.0/0 --gateway-id $WAF_IGW

# Associate with subnets
aws ec2 associate-route-table --route-table-id $WAF_RT --subnet-id $WAF_SUBNET_1
aws ec2 associate-route-table --route-table-id $WAF_RT --subnet-id $WAF_SUBNET_2

echo "     ✅ Route table configured and associated"
echo ""
echo "✅ VPC Infrastructure Complete!"
echo ""
echo "Summary:"
echo "  VPC:       $WAF_VPC (10.4.0.0/16)"
echo "  Subnet 1:  $WAF_SUBNET_1 (10.4.1.0/24) in us-east-1a"
echo "  Subnet 2:  $WAF_SUBNET_2 (10.4.2.0/24) in us-east-1b"
echo "  IGW:       $WAF_IGW"
echo "  Route Table: $WAF_RT"
echo ""
```

### **Step 2: Create Application Load Balancer (15 minutes)**

```bash
#!/bin/bash
# CloudShell - Create ALB

echo "Step 2: Creating Application Load Balancer"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo ""
echo "ALB will be the target for WAF protection"
echo "All HTTP traffic will go through WAF before reaching ALB"
echo ""

# Create security group for ALB
echo "  → Creating security group for ALB..."
WAF_ALB_SG=$(aws ec2 create-security-group \
  --group-name waf-alb-sg \
  --description "Security group for WAF-protected ALB" \
  --vpc-id $WAF_VPC \
  --tag-specifications 'ResourceType=security-group,Tags=[{Key=Name,Value=WAF-ALB-SG}]' \
  --query 'GroupId' \
  --output text)

echo "     ✅ Security group created: $WAF_ALB_SG"
echo ""
echo "     Adding inbound rules..."

# Allow HTTP from anywhere (WAF will protect)
aws ec2 authorize-security-group-ingress \
  --group-id $WAF_ALB_SG \
  --protocol tcp \
  --port 80 \
  --cidr 0.0.0.0/0

echo "       • HTTP (80) from anywhere"

# Allow HTTPS from anywhere
aws ec2 authorize-security-group-ingress \
  --group-id $WAF_ALB_SG \
  --protocol tcp \
  --port 443 \
  --cidr 0.0.0.0/0

echo "       • HTTPS (443) from anywhere"
echo ""
echo "     Note: Security group allows all HTTP/HTTPS"
echo "           WAF will provide the actual protection!"
echo ""

# Create Application Load Balancer
echo "  → Creating Application Load Balancer..."
echo "     This takes about 2-3 minutes..."
echo ""

WAF_ALB_ARN=$(aws elbv2 create-load-balancer \
  --name WAF-Protected-ALB \
  --subnets $WAF_SUBNET_1 $WAF_SUBNET_2 \
  --security-groups $WAF_ALB_SG \
  --scheme internet-facing \
  --type application \
  --ip-address-type ipv4 \
  --tags Key=Name,Value=WAF-Protected-ALB \
  --query 'LoadBalancers[0].LoadBalancerArn' \
  --output text)

echo "     ✅ ALB created: $WAF_ALB_ARN"
echo ""

# Get ALB DNS name
WAF_ALB_DNS=$(aws elbv2 describe-load-balancers \
  --load-balancer-arns $WAF_ALB_ARN \

echo "     DNS Name: $WAF_ALB_DNS"
echo "     Access via: http://$WAF_ALB_DNS"
echo ""

# Wait for ALB to be active
echo "  → Waiting for ALB to become active..."

while true; do
    STATE=$(aws elbv2 describe-load-balancers \
        --load-balancer-arns $WAF_ALB_ARN \
        --query 'LoadBalancers[0].State.Code' \
        --output text)

    if [ "$STATE" == "active" ]; then
        echo "     ✅ ALB is active!"
        break
    fi

    echo "     State: $STATE (waiting...)"
    sleep 10
done

echo ""

# Create target group
echo "  → Creating target group..."
WAF_TG_ARN=$(aws elbv2 create-target-group \
  --name WAF-Test-TG \
  --protocol HTTP \
  --port 80 \
  --vpc-id $WAF_VPC \
  --health-check-path / \
  --health-check-interval-seconds 30 \
  --health-check-timeout-seconds 5 \
  --healthy-threshold-count 2 \
  --unhealthy-threshold-count 3 \
  --tags Key=Name,Value=WAF-Test-TG \
  --query 'TargetGroups[0].TargetGroupArn' \
  --output text)

echo "     ✅ Target group created: $WAF_TG_ARN"
echo ""

# Create HTTP listener
echo "  → Creating HTTP listener..."
aws elbv2 create-listener \
  --load-balancer-arn $WAF_ALB_ARN \
  --protocol HTTP \
  --port 80 \
  --default-actions Type=forward,TargetGroupArn=$WAF_TG_ARN

echo "     ✅ HTTP listener created (port 80 → target group)"
echo ""

# Launch test EC2 instance
echo "  → Launching test web server instance..."
AMI_ID=$(aws ec2 describe-images \
  --owners amazon \
  --filters "Name=name,Values=al2023-ami-2023.*-x86_64" "Name=state,Values=available" \
  --query 'Images | sort_by(@, &CreationDate) | [-1].ImageId' \
  --output text)

echo "     Using AMI: $AMI_ID"
echo ""

# Create security group for instances
WAF_INSTANCE_SG=$(aws ec2 create-security-group \
  --group-name waf-instance-sg \
  --description "Security group for WAF test instances" \
  --vpc-id $WAF_VPC \
  --tag-specifications 'ResourceType=security-group,Tags=[{Key=Name,Value=WAF-Instance-SG}]' \
  --query 'GroupId' \
  --output text)

# Allow HTTP from ALB security group
aws ec2 authorize-security-group-ingress \
  --group-id $WAF_INSTANCE_SG \
  --protocol tcp \
  --port 80 \
  --source-group $WAF_ALB_SG

echo "     ✅ Instance security group: $WAF_INSTANCE_SG"
echo "        (Allows HTTP from ALB only)"
echo ""

# Launch instance
WAF_INSTANCE=$(aws ec2 run-instances \
  --image-id $AMI_ID \

  --user-data '#!/bin/bash

### **Step 3: Create IP Set (Blocklist) (5 minutes)**

```bash
#!/bin/bash
# CloudShell - Create IP Set

echo "Step 3: Creating IP Set (Malicious IP Blocklist)"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo ""
echo "IP Sets allow you to block or allow specific IP addresses"
echo "Useful for: Known attackers, threat intelligence feeds"
echo ""

# Create IP set with malicious IPs
echo "  → Creating IP set with sample malicious IPs..."
echo ""
echo "     IPs to block:"
echo "       • 198.51.100.0/24 (TEST-NET-2, example range)"
echo "       • 203.0.113.0/24 (TEST-NET-3, example range)"
echo "       • 192.0.2.0/24 (TEST-NET-1, example range)"
echo ""
echo "     Note: These are documentation IPs (safe for testing)"
echo "           In production, use actual threat intelligence IPs"
echo ""

WAF_IP_SET_ID=$(aws wafv2 create-ip-set \
  --name MaliciousIPBlocklist \
  --scope REGIONAL \
  --ip-address-version IPV4 \
  --addresses "198.51.100.0/24" "203.0.113.0/24" "192.0.2.0/24" \
  --description "Known malicious IP addresses - automatically blocked" \
  --tags Key=Name,Value=MaliciousIPBlocklist \
  --query 'Summary.Id' \
  --output text)

# Get IP Set ARN
WAF_IP_SET_ARN=$(aws wafv2 list-ip-sets \
  --scope REGIONAL \
  --query "IPSets[?Name=='MaliciousIPBlocklist'].ARN" \
  --output text)

echo "     ✅ IP Set created!"
echo "     ID:  $WAF_IP_SET_ID"
echo "     ARN: $WAF_IP_SET_ARN"
echo ""
echo "     Capacity: 3 IP ranges"
echo "     Maximum: 10,000 IP addresses per set"
echo ""
echo "  → How IP Sets work:"
echo "     1. Traffic arrives with source IP"
echo "     2. WAF checks if IP is in blocklist"
echo "     3. If match: Request BLOCKED (403 Forbidden)"
echo "     4. If no match: Continue to next rule"
echo ""
echo "  → Updating IP Sets:"
echo "     • Can update in real-time via API"
echo "     • Changes propagate instantly"
echo "     • Perfect for dynamic threat feeds"
echo "     • Lambda can auto-update from threat intel"
echo ""
echo "✅ IP Set Ready!"
echo ""

### **Step 4: Create Comprehensive Web ACL (20 minutes)**

```bash
#!/bin/bash
# CloudShell - Create Web ACL with All Rules

echo "Step 4: Creating Comprehensive Web ACL"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo ""
echo "Web ACL = Web Access Control List"
echo "Container for all security rules"
echo ""
echo "We'll create rules for:"
echo "  1. IP-based blocking (malicious IPs)"
echo "  2. Rate limiting (DDoS mitigation)"
echo "  3. AWS Managed - Core Rule Set (OWASP Top 10)"
echo "  4. AWS Managed - Known Bad Inputs"
echo "  5. AWS Managed - SQL Injection"
echo "  6. AWS Managed - Admin Protection"
echo "  7. Geo-blocking (optional, commented out)"
echo ""

# Create comprehensive Web ACL
cat > /tmp/waf-web-acl.json << EOF
{
  "Name": "ComprehensiveWebACL",
  "Scope": "REGIONAL",
  "DefaultAction": {
    "Allow": {}
  },
  "Description": "Comprehensive WAF protection with multiple rule types",
  "Rules": [
    {
      "Name": "BlockMaliciousIPs",
      "Priority": 0,
      "Statement": {
        "IPSetReferenceStatement": {
          "Arn": "$WAF_IP_SET_ARN"
        }
      },
      "Action": {
        "Block": {
          "CustomResponse": {
            "ResponseCode": 403,
            "CustomResponseBodyKey": "BlockedIP"
          }
        }
      },
      "VisibilityConfig": {
        "SampledRequestsEnabled": true,
        "CloudWatchMetricsEnabled": true,
        "MetricName": "BlockMaliciousIPs"
      }
    },
    {
      "Name": "RateLimitProtection",
      "Priority": 1,
      "Statement": {
        "RateBasedStatement": {
          "Limit": 2000,
          "AggregateKeyType": "IP"
        }
      },
      "Action": {
        "Block": {
          "CustomResponse": {
            "ResponseCode": 429,
            "CustomResponseBodyKey": "RateLimited"
          }
        }
      },
      "VisibilityConfig": {
        "SampledRequestsEnabled": true,
        "CloudWatchMetricsEnabled": true,
        "MetricName": "RateLimitProtection"
      }

## Components Deep Dive

### 1. Stateless Rules
- **Evaluated FIRST** (before stateful)
- **No connection tracking**
- **Priority-based** (1-65535, lower = higher priority)
- **Fast packet filtering**

**Actions:**
- `aws:pass` - Allow immediately, skip stateful
- `aws:drop` - Drop packet immediately
- `aws:forward_to_sfe` - Send to stateful engine

**Use Cases:**
- Quick DROP of known malicious IPs
- PASS trusted internal traffic (performance optimization)
- Rate limiting with custom actions

**Example Rules:**
```
Priority 1: DROP 198.51.100.0/24 (malicious IPs)
Priority 10: FORWARD port 80,443 to stateful
Priority 100: FORWARD all other to stateful
```

---

### 2. Stateful Rules (Three Types)

#### A) Standard 5-Tuple Rules

**Format:** Protocol, Source IP/Port, Dest IP/Port, Direction

**Actions:**
- `PASS` - Allow traffic
- `DROP` - Block silently
- `REJECT` - Block with TCP reset/ICMP unreachable
- `ALERT` - Log only (IDS mode)

**Example:**
```json
{
  "Action": "DROP",
  "Header": {
    "Protocol": "TCP",
    "Source": "ANY",
    "SourcePort": "ANY",
    "Direction": "FORWARD",
    "Destination": "ANY",
    "DestinationPort": "22"
  },
  "RuleOptions": [{"Keyword": "sid:1000001"}]
}
```

**Use:** Block specific ports (SSH, RDP, Telnet)

---

#### B) Domain Filtering

**Purpose:** Filter by domain name (not just IP!)

**How It Works:**
- Inspects HTTP `Host:` header
- Inspects TLS SNI (Server Name Indication)
- Supports wildcards (`.facebook.com` = all subdomains)

**Rule Types:**
- `ALLOWLIST` - Only listed domains allowed, block all else
- `DENYLIST` - Block listed domains, allow all else

**Example:**
```json
{
  "RulesSourceList": {
    "TargetTypes": ["HTTP_HOST", "TLS_SNI"],
    "Targets": [
      ".facebook.com",
      ".twitter.com",
      ".instagram.com"
    ],
    "GeneratedRulesType": "DENYLIST"
  }
}
```

**Use Cases:**
- Corporate internet filtering (block social media)
- Compliance (block gambling, adult content)
- Malware prevention (block known C2 domains)

---

#### C) Suricata IDS/IPS Rules

**Suricata** = Open-source IDS/IPS engine with deep packet inspection

**Rule Syntax:**
```
action protocol src_ip src_port direction dst_ip dst_port (options)
```

**Example Rules Created:**

```
SID 1000010: SQL Injection - "UNION SELECT"
alert http any any -> any any (
  msg:"SQL Injection Detected";
  flow:to_server,established;
  content:"UNION SELECT";
  nocase;
  sid:1000010;
  rev:1;
)

SID 1000012: XSS - "<script>" tag
alert http any any -> any any (
  msg:"XSS Attack Detected";
  flow:to_server,established;
  content:"<script>";
  nocase;
  sid:1000012;
  rev:1;
)

SID 1000014: Path Traversal - "../"
alert http any any -> any any (
  msg:"Path Traversal Attack";
  flow:to_server,established;
  content:"..%2f";
  nocase;
  sid:1000014;
  rev:1;
)
```

**Rule Options:**
- `msg` - Alert message
- `flow` - Connection state
- `content` - Pattern to match
- `pcre` - Regex pattern
- `sid` - Signature ID (must be unique)
- `nocase` - Case-insensitive matching

**IDS vs IPS Mode:**
- `alert` - IDS mode (log only, don't block)
- `drop` - IPS mode (block traffic)

**Best Practice:**
1. Start with `alert` (IDS)
2. Monitor logs for false positives
3. Tune rules
4. Switch to `drop` (IPS) after validation

## Traffic Flow

**Inbound (Internet → EC2):**
```
1. Packet arrives at Internet Gateway
2. IGW routes to firewall endpoint (via route table)
3. Firewall evaluates:
   a. Stateless rules (quick filter)
   b. Stateful rules (deep inspection)
   c. Domain filtering
   d. Suricata IDS
4. Decision: ALLOW or DROP
5. If ALLOW: Forwarded to protected subnet
6. EC2 receives packet
```

**Outbound (EC2 → Internet):**
```
1. EC2 sends packet
2. Protected subnet RT routes to firewall endpoint
3. Firewall inspects (same as inbound)
4. If ALLOW: Forwarded to IGW
5. IGW sends to internet
```

**ALL traffic inspected in BOTH directions!** 🔍

## Exam Tips

**What to Know:**

1. **Stateless vs Stateful Difference:**
   - Stateless: No connection tracking, both directions need rules, priority-based
   - Stateful: Connection tracking, automatic return traffic, actions include PASS/DROP/REJECT/ALERT

2. **When to Use Network Firewall vs Others:**
   - Security Groups: Instance-level, stateful, simple allow rules
   - NACLs: Subnet-level, stateless, allow + deny rules
   - Network Firewall: VPC-level, IDS/IPS, domain filtering, deep inspection
   - WAF: Application-layer (HTTP/HTTPS), web attacks, rate limiting

3. **Domain Filtering Limitations:**
   - HTTP/HTTPS only
   - Requires Host header or SNI
   - Encrypted SNI (ESNI) not supported

4. **Suricata IDS vs IPS:**
   - IDS: `alert` action, logs only
   - IPS: `drop` action, blocks traffic
   - Start with IDS, switch to IPS after tuning

5. **Cost Components:**
   - Endpoint: $0.395/hour/AZ (fixed)
   - Data: $0.065/GB (variable)
   - Minimum 2 AZ for HA

6. **Capacity Units:**
   - Set at creation, CANNOT be changed
   - Range: 100 to 30,000 per rule group
   - Plan ahead!

❓ *"You need to block access to social media sites. Which service?"*  
✅ Network Firewall (domain filtering) OR Web Application Firewall (string matching)

❓ *"PCI-DSS requires IDS/IPS. Which service?"*

✅ Security Group (simple, free) OR Network Firewall if already deployed

**Protects:**

## WAF Components

### 1. Web ACL (Access Control List)

**Container for all rules**

**Properties:**
- Name: Unique identifier
- Scope: REGIONAL (ALB, API Gateway) or CLOUDFRONT (global)
- Default Action: ALLOW or BLOCK
- Rules: Priority-ordered list
- Capacity: Max 1500 WCU (Web ACL Capacity Units)

**Default Action Strategy:**

**Approach 1: Default ALLOW** (Most Common)
- Allow everything EXCEPT what's blocked
- Rules define blocklist
- Good for: Public websites
- Example: Block only known attacks

**Approach 2: Default BLOCK** (Strict)
- Block everything EXCEPT what's allowed
- Rules define allowlist
- Good for: High-security APIs
- Example: Allow only specific patterns

---

### 2. Rules

**Three Types:**

#### A) Regular Rules
- Custom conditions you define
- Match statements (what to look for)
- Action (what to do if matched)

#### B) Rate-Based Rules
- Count requests per 5-minute window
- Threshold: 100 to 2,000,000,000
- Scope: Per IP or custom keys
- Action: Block automatically when exceeded

#### C) Managed Rule Groups
- Pre-built by AWS
- Maintained and updated automatically
- Cover OWASP Top 10
- Most are FREE!

---

### 3. Rule Actions

| Action | Description | HTTP Code |
|--------|-------------|-----------|
| ALLOW | Let request through | 200 (pass-through) |
| BLOCK | Block request | 403 Forbidden |
| COUNT | Count only (monitor) | 200 (pass-through) |
| CAPTCHA | Challenge with CAPTCHA | CAPTCHA page |
| Challenge | JavaScript challenge | JS challenge |

**Best Practice Workflow:**
1. Deploy in COUNT mode
2. Monitor for false positives (1-2 weeks)
3. Tune rules
4. Switch to CHALLENGE mode
5. Monitor user complaints
6. If acceptable, switch to BLOCK

---

### 4. Match Statements

**Available Match Types:**

**Geographic Match:**
- Match by country code (ISO 3166-1 alpha-2)
- Example: Block CN, RU, NK
- Accuracy: ~95-99%
- Limitation: VPN/proxy bypass

**IP Set Match:**
- Match against IP lists
- Up to 10,000 IPs per set
- Dynamic updates via API
- Use: Blocklist/allowlist

**Regex Pattern Set:**
- Regular expression matching
- Up to 10 patterns per set
- Use: Complex pattern detection
- Example: Credit card numbers

**Size Constraint:**
- Check request component size
- Operators: EQ, NE, GT, GE, LT, LE
- Use: Limit request body size

**SQL Injection Match:**
- AWS-managed signatures
- Detects SQL injection patterns
- Example: `UNION SELECT`

**XSS Match:**
- AWS-managed signatures
- Detects cross-site scripting
- Example: `<script>` tags

**String Match:**
- Simple string matching
- Case-sensitive or insensitive
- Position: EXACTLY, STARTS_WITH, ENDS_WITH, CONTAINS
- Transformations: LOWERCASE, URL_DECODE, HTML_DECODE

## Rate-Based Rules Deep Dive

**How They Work:**
1. Track requests per IP (or custom scope)
2. Count over 5-minute rolling window
3. If count > threshold: Take action
4. Block duration: ~10 minutes (auto-expires)

**Configuration:**

**Threshold:**
- Minimum: 100 requests/5 min
- Maximum: 2,000,000,000 requests/5 min
- Typical: 2000-10000

**Scope Options:**
- `IP` - Per source IP (most common)
- `FORWARDED_IP` - Per X-Forwarded-For header
- `CUSTOM_KEYS` - Custom aggregation

**Common Thresholds:**

| Use Case | Threshold | Reason |
|----------|-----------|--------|
| DDoS Mitigation | 10,000 | Normal user won't send 10k in 5 min |
| Login Protection | 20 | Prevent brute force |
| API Rate Limit | 1,000 | Fair usage |
| Search Throttling | 500 | Expensive operation |
| Registration | 5 | Prevent fake accounts |

## Key Learnings

**Technical:**
1. **WAF operates at Layer 7** - Understands HTTP/HTTPS
2. **Managed Rules are FREE** - Use them! No reason not to
3. **Rate-based rules are powerful** - Simple DDoS mitigation
4. **CAPTCHA vs Challenge** - CAPTCHA = visible, Challenge = invisible
5. **Geographic blocking has limits** - VPN/proxy bypass

**Operational:**
1. **Start with COUNT mode** - Prevent breaking legitimate traffic
2. **One Web ACL protects multiple resources** - Cost-effective
3. **Monitor sampled requests** - Understand traffic patterns
4. **Custom response bodies help users** - Better than generic 403
5. **Combine with Shield Standard** - FREE DDoS protection

**Cost:**
1. **Very affordable** - $10-50/month typical
2. **Managed rules are FREE** - Don't write custom rules unless needed
3. **Bot Control adds up** - Only if you have bot problem
4. **WAF + Shield Standard** - Great value ($15/month)

## Exam Tips

**What to Know:**

1. **WAF vs Network Firewall:**
   - WAF: Layer 7, web attacks, rate limiting, $10-50/month
   - Network Firewall: Layer 3-7, IDS/IPS, domain filtering, $600+/month

2. **Shield Standard vs Advanced:**
   - Standard: FREE, Layer 3/4, automatic, 99.99% effective
   - Advanced: $3k/month, Layer 7, DRT, cost protection, EC2 EIP

3. **AWS Managed Rules:**
   - Core Rule Set: OWASP Top 10 (FREE)
   - Bot Control: Only PAID managed rule ($10/month)
   - All others: FREE

4. **Rate-Based Rules:**
   - Threshold: 100 to 2 billion req/5min
   - Scope: IP, Forwarded IP, Custom Keys
   - Block duration: ~10 minutes

5. **Rule Actions:**
   - ALLOW: Let through
   - BLOCK: 403 Forbidden
   - COUNT: Monitor only
   - CAPTCHA: Challenge user
   - Challenge: JavaScript check

**Common Exam Questions:**

❓ *"Protect web app from SQL injection. Which service?"*  
✅ AWS WAF (Core Rule Set or SQLi Rule Set)

❓ *"Block traffic from specific countries"*  
✅ AWS WAF (Geographic match statement)

❓ *"Prevent DDoS on CloudFront"*  
✅ Shield Standard (automatic, FREE) + WAF rate limiting

❓ *"Gaming company needs DDoS protection with 24/7 support"*  
✅ Shield Advanced ($3k/month)

❓ *"Block all IPs except corporate office"*  
✅ WAF with IP Set (allowlist) + default BLOCK

❓ *"Protect against OWASP Top 10"*  
✅ AWS Managed Rules Core Rule Set (FREE)

## Commands Reference

**Create IP Set:**
```bash
aws wafv2 create-ip-set \
  --name MaliciousIPs \
  --scope REGIONAL \
  --ip-address-version IPV4 \
  --addresses "198.51.100.0/24" "203.0.113.0/24"
```

**Create Web ACL:**
```bash
aws wafv2 create-web-acl \
  --name MyWebACL \
  --scope REGIONAL \
  --default-action Allow={} \
  --rules file://rules.json \
  --visibility-config SampledRequestsEnabled=true,CloudWatchMetricsEnabled=true,MetricName=MyWebACL
```

**Associate with ALB:**
```bash
aws wafv2 associate-web-acl \
  --web-acl-arn arn:aws:wafv2:... \
  --resource-arn arn:aws:elasticloadbalancing:...
```

**View Sampled Requests:**
```bash
aws wafv2 get-sampled-requests \
  --web-acl-arn arn:... \
  --rule-metric-name RuleName \
  --scope REGIONAL \
  --time-window StartTime=...,EndTime=... \
  --max-items 100
```
