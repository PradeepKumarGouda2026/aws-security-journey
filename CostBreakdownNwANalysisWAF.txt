#!/bin/bash

## **IF YOU WANT TO DELETE EVERYTHING (CLEANUP)**

```bash
#!/bin/bash
# Cleanup Network Firewall Lab (when you don't have variables)

echo "╔═══════════════════════════════════════════════════════════╗"
echo "║    NETWORK FIREWALL LAB 88 - COMPLETE CLEANUP           ║"
echo "╚═══════════════════════════════════════════════════════════╝"
echo ""

echo "⚠️  This will delete ALL Network Firewall resources!"
echo "Press Ctrl+C to cancel, or Enter to continue..."
read

# Step 1: Delete all firewalls
echo "Step 1: Deleting firewalls..."
for FIREWALL_ARN in $(aws network-firewall list-firewalls --query 'Firewalls[*].FirewallArn' --output text); do
    FIREWALL_NAME=$(echo $FIREWALL_ARN | awk -F'/' '{print $NF}')
    echo "  Deleting firewall: $FIREWALL_NAME"
    aws network-firewall delete-firewall --firewall-name $FIREWALL_NAME 2>/dev/null
done

echo "  Waiting for firewall deletion (60 seconds)..."
sleep 60

# Step 2: Delete all firewall policies
echo ""
echo "Step 2: Deleting firewall policies..."
for POLICY_ARN in $(aws network-firewall list-firewall-policies --query 'FirewallPolicies[*].Arn' --output text); do
    echo "  Deleting policy: $POLICY_ARN"
    aws network-firewall delete-firewall-policy --firewall-policy-arn $POLICY_ARN 2>/dev/null
done

# Step 3: Delete all rule groups
echo ""
echo "Step 3: Deleting rule groups..."
for RULE_GROUP_ARN in $(aws network-firewall list-rule-groups --query 'RuleGroups[*].Arn' --output text); do
    RULE_GROUP_NAME=$(echo $RULE_GROUP_ARN | awk -F'/' '{print $NF}')
    echo "  Deleting rule group: $RULE_GROUP_NAME"
    aws network-firewall delete-rule-group --rule-group-arn $RULE_GROUP_ARN 2>/dev/null
done

# Step 4: Terminate EC2 instances
echo ""
echo "Step 4: Terminating EC2 instances..."
for INSTANCE in $(aws ec2 describe-instances \
  --filters "Name=tag:Name,Values=NFW_TEST_INSTANCE" "Name=instance-state-name,Values=running,stopped" \
  --query 'Reservations[*].Instances[*].InstanceId' \
  --output text); do
    echo "  Terminating instance: $INSTANCE"
    aws ec2 terminate-instances --instance-ids $INSTANCE
done

echo "  Waiting for instance termination (30 seconds)..."
sleep 30

# Step 5: Delete VPCs with tag "Network-Firewall-VPC"
echo ""
echo "Step 5: Deleting VPC..."
for VPC in $(aws ec2 describe-vpcs \
  --filters "Name=tag:Name,Values=Network-Firewall-VPC" \
  --query 'Vpcs[*].VpcId' \
  --output text); do
    
    echo "  Found VPC: $VPC"
    
    # Delete security groups
    for SG in $(aws ec2 describe-security-groups --filters "Name=vpc-id,Values=$VPC" --query 'SecurityGroups[?GroupName!=`default`].GroupId' --output text); do
        echo "    Deleting security group: $SG"
        aws ec2 delete-security-group --group-id $SG 2>/dev/null
    done
    
    # Detach and delete IGW
    for IGW in $(aws ec2 describe-internet-gateways --filters "Name=attachment.vpc-id,Values=$VPC" --query 'InternetGateways[*].InternetGatewayId' --output text); do
        echo "    Detaching IGW: $IGW"
        aws ec2 detach-internet-gateway --internet-gateway-id $IGW --vpc-id $VPC
        aws ec2 delete-internet-gateway --internet-gateway-id $IGW
    done
    
    # Delete route tables
    for RT in $(aws ec2 describe-route-tables --filters "Name=vpc-id,Values=$VPC" --query 'RouteTables[?Associations[0].Main!=`true`].RouteTableId' --output text); do
        # Disassociate first
        for ASSOC in $(aws ec2 describe-route-tables --route-table-ids $RT --query 'RouteTables[0].Associations[?SubnetId!=`null`].RouteTableAssociationId' --output text); do
            echo "    Disassociating route table: $RT"
            aws ec2 disassociate-route-table --association-id $ASSOC 2>/dev/null
        done
        echo "    Deleting route table: $RT"
        aws ec2 delete-route-table --route-table-id $RT 2>/dev/null
    done
    
    # Delete subnets
    for SUBNET in $(aws ec2 describe-subnets --filters "Name=vpc-id,Values=$VPC" --query 'Subnets[*].SubnetId' --output text); do
        echo "    Deleting subnet: $SUBNET"
        aws ec2 delete-subnet --subnet-id $SUBNET
    done
    
    # Delete VPC
    echo "    Deleting VPC: $VPC"
    aws ec2 delete-vpc --vpc-id $VPC
done

echo ""
echo "╔═══════════════════════════════════════════════════════════╗"
echo "║  CLEANUP COMPLETE! ✅                                    ║"
echo "╚═══════════════════════════════════════════════════════════╝"
echo ""
echo "All Network Firewall Lab 88 resources have been deleted."
echo ""
```
