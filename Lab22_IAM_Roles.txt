## **COMPLETE LAB SCRIPT**

```bash
#!/bin/bash
# Lab 95: Cross-Account Access with IAM Roles
# Complete working script with all scenarios

set -e  # Exit on error

echo "╔═══════════════════════════════════════════════════════════╗"
echo "║     LAB 95: CROSS-ACCOUNT ACCESS WITH IAM ROLES         ║"
echo "╚═══════════════════════════════════════════════════════════╝"
echo ""

#############################################################################
# INTRODUCTION
#############################################################################

cat << 'EOF'
SCENARIO:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

This lab simulates cross-account access:

"Account A" (Development):
├─ Represented by: IAM User 'developer'
├─ Developers work here
└─ No direct production access

"Account B" (Production):
├─ Represented by: IAM Role 'ProductionAccessRole'
├─ Production resources here
└─ Protected with strict access controls

Goal: Developer needs temporary read-only access to production

In real enterprise:
- Developer would be in Account 111111111111
- Production role would be in Account 222222222222
- Same concepts, different account IDs!

Security measures:
✅ External ID (prevents confused deputy attack)
✅ Temporary credentials (1 hour, auto-expire)
✅ Least privilege (read-only)
✅ Session policies (further restrictions)
✅ Full audit trail

EOF

read -p "Press Enter to start lab..."

#############################################################################
# STEP 1: CREATE PRODUCTION S3 BUCKET
#############################################################################

echo ""
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "STEP 1: Creating Production S3 Bucket"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo ""

TIMESTAMP=$(date +%s)
PROD_BUCKET="production-logs-${TIMESTAMP}"
DEV_BUCKET="development-data-${TIMESTAMP}"

# Create production bucket
aws s3 mb s3://${PROD_BUCKET}
echo "✅ Production bucket created: ${PROD_BUCKET}"

# Create development bucket (for comparison)
aws s3 mb s3://${DEV_BUCKET}
echo "✅ Development bucket created: ${DEV_BUCKET}"

# Upload test files
echo "Production sensitive data - Q4 2025 financials" > /tmp/prod-data.txt
echo "Development test data - sample records" > /tmp/dev-data.txt

aws s3 cp /tmp/prod-data.txt s3://${PROD_BUCKET}/sensitive/financials-q4.txt
aws s3 cp /tmp/dev-data.txt s3://${DEV_BUCKET}/test/sample.txt

echo "✅ Test files uploaded to both buckets"

# Get account ID
ACCOUNT_ID=$(aws sts get-caller-identity --query Account --output text)
echo "✅ Account ID: ${ACCOUNT_ID}"

echo ""
echo "Resources created:"
echo "  Production: s3://${PROD_BUCKET}/"
echo "  Development: s3://${DEV_BUCKET}/"
echo ""

#############################################################################
# STEP 2: CREATE DEVELOPER USER (Represents Account A)
#############################################################################

echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "STEP 2: Creating Developer User"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo ""

# Create developer user
aws iam create-user --user-name developer 2>/dev/null || echo "User already exists"
echo "✅ User 'developer' created (represents Account A)"

# Give developer access to development bucket only
cat > /tmp/dev-permissions.json << EOF
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Sid": "AllowDevBucketAccess",
      "Effect": "Allow",
      "Action": [
        "s3:GetObject",
        "s3:PutObject",
        "s3:ListBucket"
      ],
      "Resource": [
        "arn:aws:s3:::${DEV_BUCKET}",
        "arn:aws:s3:::${DEV_BUCKET}/*"
      ]
    },
    {
      "Sid": "AllowListAllBuckets",
      "Effect": "Allow",
      "Action": "s3:ListAllMyBuckets",
      "Resource": "*"
    }
  ]
}
EOF

aws iam put-user-policy \
  --user-name developer \
  --policy-name DevBucketAccess \
  --policy-document file:///tmp/dev-permissions.json

echo "✅ Developer has access to development bucket only"

# Create access keys for developer
echo ""
echo "Creating access keys for developer..."
KEYS_JSON=$(aws iam create-access-key --user-name developer 2>/dev/null || \
            aws iam list-access-keys --user-name developer --query 'AccessKeyMetadata[0].AccessKeyId' --output text)

if [[ $KEYS_JSON == *"AccessKey"* ]]; then
    DEV_ACCESS_KEY=$(echo $KEYS_JSON | grep -o '"AccessKeyId":"[^"]*"' | cut -d'"' -f4)
    DEV_SECRET_KEY=$(echo $KEYS_JSON | grep -o '"SecretAccessKey":"[^"]*"' | cut -d'"' -f4)
    echo "✅ New access keys created"
else
    # Keys already exist, use existing
    DEV_ACCESS_KEY=$(aws iam list-access-keys --user-name developer --query 'AccessKeyMetadata[0].AccessKeyId' --output text)
    echo "✅ Using existing access keys"
    echo ""
    echo "⚠️  Note: Secret key not available for existing keys"
    echo "    For full lab functionality, delete old keys and run again"
    echo ""
    DEV_SECRET_KEY="<existing-key-secret-not-retrievable>"
fi

echo ""
echo "Developer credentials (Account A simulation):"
echo "  Access Key: ${DEV_ACCESS_KEY}"
echo "  Secret Key: ${DEV_SECRET_KEY:0:20}..."
echo ""

#############################################################################
# STEP 3: CREATE PRODUCTION ACCESS ROLE (Represents Account B)
#############################################################################

echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "STEP 3: Creating Production Access Role"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo ""

# Generate unique External ID
EXTERNAL_ID="MyCompany-DevToProd-$(uuidgen 2>/dev/null || echo $(date +%s)-RANDOM-$$)"
echo "External ID: ${EXTERNAL_ID}"
echo "⚠️  SAVE THIS! You'll need it to assume the role"
echo ""

# Create trust policy
cat > /tmp/trust-policy.json << EOF
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Sid": "AllowDeveloperToAssume",
      "Effect": "Allow",
      "Principal": {
        "AWS": "arn:aws:iam::${ACCOUNT_ID}:user/developer"
      },
      "Action": "sts:AssumeRole",
      "Condition": {
        "StringEquals": {
          "sts:ExternalId": "${EXTERNAL_ID}"
        }
      }
    }
  ]
}
EOF

echo "Trust Policy (WHO can assume this role):"
cat /tmp/trust-policy.json
echo ""

# Create role
aws iam create-role \
  --role-name ProductionAccessRole \
  --assume-role-policy-document file:///tmp/trust-policy.json \
  --description "Cross-account read-only access to production resources" \
  --max-session-duration 3600 \
  2>/dev/null || echo "Role already exists, updating trust policy..."

# Update trust policy if role exists
aws iam update-assume-role-policy \
  --role-name ProductionAccessRole \
  --policy-document file:///tmp/trust-policy.json \
  2>/dev/null || true

ROLE_ARN="arn:aws:iam::${ACCOUNT_ID}:role/ProductionAccessRole"
echo "✅ Role created: ProductionAccessRole"
echo "   ARN: ${ROLE_ARN}"
echo ""

# Create permissions policy (WHAT the role can do)
cat > /tmp/permissions-policy.json << EOF
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Sid": "ReadProductionS3",
      "Effect": "Allow",
      "Action": [
        "s3:GetObject",
        "s3:ListBucket",
        "s3:GetBucketLocation"
      ],
      "Resource": [
        "arn:aws:s3:::${PROD_BUCKET}",
        "arn:aws:s3:::${PROD_BUCKET}/*"
      ]
    },
    {
      "Sid": "ListAllBuckets",
      "Effect": "Allow",
      "Action": "s3:ListAllMyBuckets",
      "Resource": "*"
    },
    {
      "Sid": "DenyDestructiveActions",
      "Effect": "Deny",
      "Action": [
        "s3:DeleteObject",
        "s3:DeleteBucket",
        "s3:PutObject"
      ],
      "Resource": "*"
    }
  ]
}
EOF

echo "Permissions Policy (WHAT the role can do):"
cat /tmp/permissions-policy.json
echo ""

# Attach permissions policy
aws iam put-role-policy \
  --role-name ProductionAccessRole \
  --policy-name ProductionReadOnlyAccess \
  --policy-document file:///tmp/permissions-policy.json

echo "✅ Permissions policy attached"
echo "   - CAN: Read production S3 bucket"
echo "   - CANNOT: Write, delete, or modify"
echo ""

#############################################################################
# STEP 4: GRANT DEVELOPER PERMISSION TO ASSUME ROLE
#############################################################################

echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "STEP 4: Grant Developer Permission to Assume Role"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo ""

# Create policy allowing developer to assume the role
cat > /tmp/assume-role-policy.json << EOF
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Sid": "AllowAssumeProductionRole",
      "Effect": "Allow",
      "Action": "sts:AssumeRole",
      "Resource": "${ROLE_ARN}"
    }
  ]
}
EOF

aws iam put-user-policy \
  --user-name developer \
  --policy-name AssumeProductionRole \
  --policy-document file:///tmp/assume-role-policy.json

echo "✅ Developer can now assume ProductionAccessRole"
echo ""

#############################################################################
# SAVE CURRENT CREDENTIALS
#############################################################################

echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "SAVING CURRENT ADMIN CREDENTIALS"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo ""

# Save current credentials (admin)
ADMIN_ACCESS_KEY=$AWS_ACCESS_KEY_ID
ADMIN_SECRET_KEY=$AWS_SECRET_ACCESS_KEY
ADMIN_SESSION_TOKEN=$AWS_SESSION_TOKEN

echo "✅ Admin credentials saved"
echo ""

#############################################################################
# STEP 5: TEST - DEVELOPER DIRECT ACCESS (SHOULD FAIL)
#############################################################################

echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "STEP 5: Test Direct Access to Production (Should FAIL)"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo ""

if [ "$DEV_SECRET_KEY" != "<existing-key-secret-not-retrievable>" ]; then
    echo "Switching to developer credentials..."
    export AWS_ACCESS_KEY_ID=$DEV_ACCESS_KEY
    export AWS_SECRET_ACCESS_KEY=$DEV_SECRET_KEY
    unset AWS_SESSION_TOKEN
    
    echo ""
    echo "Test 1: Can developer list all buckets?"
    aws s3 ls 2>&1 | head -5
    echo "✅ Can list buckets (basic permission)"
    echo ""
    
    echo "Test 2: Can developer access development bucket?"
    echo "Command: aws s3 ls s3://${DEV_BUCKET}/"
    aws s3 ls s3://${DEV_BUCKET}/ 2>&1
    echo "✅ SUCCESS: Developer can access development bucket"
    echo ""
    
    echo "Test 3: Can developer access production bucket directly?"
    echo "Command: aws s3 ls s3://${PROD_BUCKET}/"
    aws s3 ls s3://${PROD_BUCKET}/ 2>&1 || echo "❌ ACCESS DENIED (Expected! Developer has no direct prod access)"
    echo ""
    
    echo "✅ Verification complete:"
    echo "   - Developer CAN access dev bucket ✓"
    echo "   - Developer CANNOT access prod bucket ✗ (needs to assume role)"
    echo ""
else
    echo "⚠️  Skipping direct access tests (using existing access keys)"
    echo "   To run full tests, delete existing keys and re-run lab"
    echo ""
fi

#############################################################################
# STEP 6: ASSUME ROLE
#############################################################################

echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "STEP 6: Developer Assumes Production Access Role"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo ""

# Switch back to admin credentials for assume-role
export AWS_ACCESS_KEY_ID=$ADMIN_ACCESS_KEY
export AWS_SECRET_ACCESS_KEY=$ADMIN_SECRET_KEY
export AWS_SESSION_TOKEN=$ADMIN_SESSION_TOKEN

echo "Assuming role as developer (simulated)..."
echo ""
echo "Command:"
echo "aws sts assume-role \\"
echo "  --role-arn ${ROLE_ARN} \\"
echo "  --role-session-name developer-troubleshoot \\"
echo "  --external-id ${EXTERNAL_ID} \\"
echo "  --duration-seconds 3600"
echo ""

# Assume role
ASSUME_OUTPUT=$(aws sts assume-role \
  --role-arn ${ROLE_ARN} \
  --role-session-name developer-troubleshoot-$(date +%Y%m%d-%H%M%S) \
  --external-id "${EXTERNAL_ID}" \
  --duration-seconds 3600 \
  --output json)

# Extract temporary credentials
TEMP_ACCESS_KEY=$(echo $ASSUME_OUTPUT | grep -o '"AccessKeyId":"[^"]*"' | cut -d'"' -f4)
TEMP_SECRET_KEY=$(echo $ASSUME_OUTPUT | grep -o '"SecretAccessKey":"[^"]*"' | cut -d'"' -f4)
TEMP_SESSION_TOKEN=$(echo $ASSUME_OUTPUT | grep -o '"SessionToken":"[^"]*"' | cut -d'"' -f4)
EXPIRATION=$(echo $ASSUME_OUTPUT | grep -o '"Expiration":"[^"]*"' | cut -d'"' -f4)

echo "✅ Role assumed successfully!"
echo ""
echo "Temporary Credentials:"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "AccessKeyId:     ${TEMP_ACCESS_KEY}"
echo "SecretAccessKey: ${TEMP_SECRET_KEY:0:20}..."
echo "SessionToken:    ${TEMP_SESSION_TOKEN:0:30}..."
echo "Expiration:      ${EXPIRATION}"
echo ""
echo "Note: AccessKeyId starts with 'ASIA' (temporary credentials)"
echo "      Permanent keys start with 'AKIA'"
echo ""

# Use temporary credentials
export AWS_ACCESS_KEY_ID=$TEMP_ACCESS_KEY
export AWS_SECRET_ACCESS_KEY=$TEMP_SECRET_KEY
export AWS_SESSION_TOKEN=$TEMP_SESSION_TOKEN

#############################################################################
# STEP 7: VERIFY IDENTITY
#############################################################################

echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "STEP 7: Verify Assumed Role Identity"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo ""

echo "Command: aws sts get-caller-identity"
echo ""
aws sts get-caller-identity
echo ""

IDENTITY_ARN=$(aws sts get-caller-identity --query Arn --output text)
echo "Current Identity: ${IDENTITY_ARN}"
echo ""
echo "✅ Identity shows 'assumed-role/ProductionAccessRole'"
echo "   This confirms we're operating with role permissions"
echo ""

#############################################################################
# STEP 8: TEST ACCESS WITH ASSUMED ROLE
#############################################################################

echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "STEP 8: Test Access with Assumed Role (Should SUCCEED)"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo ""

echo "Test 1: List production bucket contents"
echo "Command: aws s3 ls s3://${PROD_BUCKET}/"
echo ""
aws s3 ls s3://${PROD_BUCKET}/ --recursive
echo ""
echo "✅ SUCCESS! Can now access production bucket via assumed role"
echo ""

echo "Test 2: Download production file"
echo "Command: aws s3 cp s3://${PROD_BUCKET}/sensitive/financials-q4.txt /tmp/"
echo ""
aws s3 cp s3://${PROD_BUCKET}/sensitive/financials-q4.txt /tmp/downloaded-prod.txt
echo ""
echo "File contents:"
cat /tmp/downloaded-prod.txt
echo ""
echo "✅ SUCCESS! Downloaded production file"
echo ""

echo "Test 3: Try to upload file (Should FAIL - read-only role)"
echo "Command: aws s3 cp /tmp/test.txt s3://${PROD_BUCKET}/test.txt"
echo "test data" > /tmp/test.txt
echo ""
aws s3 cp /tmp/test.txt s3://${PROD_BUCKET}/test.txt 2>&1 || echo ""
echo "❌ DENIED (Expected! Role has explicit deny on PutObject)"
echo ""

echo "Test 4: Try to delete file (Should FAIL)"
echo "Command: aws s3 rm s3://${PROD_BUCKET}/sensitive/financials-q4.txt"
echo ""
aws s3 rm s3://${PROD_BUCKET}/sensitive/financials-q4.txt 2>&1 || echo ""
echo "❌ DENIED (Expected! Role has explicit deny on DeleteObject)"
echo ""

#############################################################################
# STEP 9: SESSION POLICY (ADDITIONAL RESTRICTIONS)
#############################################################################

echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "STEP 9: Testing Session Policy (Further Restrictions)"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo ""

# Restore admin credentials
export AWS_ACCESS_KEY_ID=$ADMIN_ACCESS_KEY
export AWS_SECRET_ACCESS_KEY=$ADMIN_SECRET_KEY
export AWS_SESSION_TOKEN=$ADMIN_SESSION_TOKEN

# Create session policy (only allow access to specific folder)
cat > /tmp/session-policy.json << EOF
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Effect": "Allow",
      "Action": "s3:GetObject",
      "Resource": "arn:aws:s3:::${PROD_BUCKET}/sensitive/*"
```
